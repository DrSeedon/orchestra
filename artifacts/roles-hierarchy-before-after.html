Migration SQL

ALTER TABLE sessions ADD COLUMN role TEXT DEFAULT 'worker';
ALTER TABLE sessions ADD COLUMN parent_id TEXT REFERENCES sessions(name);
ALTER TABLE sessions ADD COLUMN parent_name TEXT;

-- Migrate existing data
UPDATE sessions SET role = 'orchestrator'
  WHERE is_orchestrator = 1;
UPDATE sessions SET role = 'worker'
  WHERE is_orchestrator = 0;

-- is_orchestrator becomes a computed view

Формула промпта (новая)

  base.md
  + roles/{session.role}.md        // динамический выбор

  + session.system_prompt           // кастомный
  + parent_context                  // инфо о parent agent

Пример frontmatter reviewer.md

---
name: reviewer
label: Code Reviewer
model: sonnet
can_spawn: false       # не может создавать агентов
auto_report_to: parent # отчёт к parent, не root
permissions:
  - read_worktree      # читает чужие worktree
  - comment_task       # комментирует задачи
skills: [codex-review]
---

auto-report flow

# СЕЙЧАС:
content-writer → PM-orchestrator  # всегда к root

# ПОСЛЕ:
content-writer → SEO-lead         # к parent

ranking-monitor → SEO-lead        # watcher тоже
