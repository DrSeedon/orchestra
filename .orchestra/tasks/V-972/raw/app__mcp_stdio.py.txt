   69	def publish_mcp_identity_canary(environ: dict[str, str] | None = None) -> Path | None:
   70	    """Prove this process is Orchestra's mcp_stdio, not a same-name impostor.
   71	
   72	    Returns the proof path, or None when no canary was planted (non-Grok launch).
   73	    """
   74	    env = os.environ if environ is None else environ
   75	    nonce = str(env.get(ORCHESTRA_MCP_CANARY_ENV) or "").strip()
   76	    if not nonce:
   77	        return None
   78	    if not _ORCHESTRA_MCP_CANARY_RE.fullmatch(nonce):
   79	        raise RuntimeError("ORCHESTRA_MCP_CANARY is not a 32-char hex nonce")
   80	    path = mcp_canary_proof_path(nonce)
   81	    tmp = path.with_name(f"{path.name}.{os.getpid()}.tmp")
   82	    try:
   83	        tmp.write_text(nonce, encoding="utf-8")
   84	        tmp.replace(path)
   85	    finally:
   86	        tmp.unlink(missing_ok=True)
   87	    return path

  109	@dataclass
  110	class ApiToolError(RuntimeError):
  111	    code: str
  112	    message: str
  113	    status: int | None = None
  114	    retryable: bool = False
  115	    request_id: str | None = None
  116	    retry_after_seconds: int | float | None = None
  117	    outcome_unknown: bool = False
  118	    details: dict[str, Any] = field(default_factory=dict)
  119	    result: Any = None
  120	
  121	    def __post_init__(self) -> None:
  122	        self.message = self.message.strip() or self.code or "ApiToolError"
  123	        RuntimeError.__init__(self, self.message)
  124	
  125	    def envelope(self) -> dict[str, Any]:
  126	        return {
  127	            "code": self.code,
  128	            "message": self.message,
  129	            "status": self.status,
  130	            "retryable": self.retryable,
  131	            "request_id": self.request_id,
  132	            "retry_after_seconds": self.retry_after_seconds,
  133	            "outcome_unknown": self.outcome_unknown,
  134	            "details": self.details,
  135	        }

  138	def _canonical_error(error: ApiToolError | dict[str, Any]) -> dict[str, Any]:
  139	    source = error.envelope() if isinstance(error, ApiToolError) else error
  140	    details = source.get("details")
  141	    if not isinstance(details, dict):
  142	        details = {"value": details}
  143	    code = str(source.get("code") or "tool_error")
  144	    message = _safe_response_text(str(source.get("message") or code).strip() or code)
  145	    outcome_unknown = bool(source.get("outcome_unknown", False))
  146	    return {
  147	        "code": code,
  148	        "message": message,
  149	        "status": source.get("status") if isinstance(source.get("status"), int) else None,
  150	        "retryable": bool(source.get("retryable", False)) and not outcome_unknown,
  151	        "request_id": str(source["request_id"]) if source.get("request_id") else None,
  152	        "retry_after_seconds": _retry_after_seconds(source.get("retry_after_seconds")),
  153	        "outcome_unknown": outcome_unknown,
  154	        "details": _safe_detail(details),
  155	    }

  158	def mcp_tool_result(
  159	    result: Any = None,
  160	    *,
  161	    error: ApiToolError | dict[str, Any] | None = None,
  162	    is_error: bool = False,
  163	    text: str = "",
  164	) -> CallToolResult:
  165	    """Build the shared shape; partial domain results may carry error with isError false."""
  166	    if is_error and error is None:
  167	        raise ValueError("is_error=True requires an error envelope")
  168	    envelope = _canonical_error(error) if error is not None else None
  169	    if not text:
  170	        if envelope:
  171	            text = f"{envelope['code']}: {envelope['message']}"
  172	        elif isinstance(result, str):
  173	            text = result
  174	        else:
  175	            text = json.dumps(result, ensure_ascii=False)
  176	    if envelope:
  177	        text = _safe_response_text(text)
  178	    return CallToolResult(
  179	        content=[TextContent(type="text", text=text or "OK")],
  180	        structuredContent={"result": result, "error": envelope},
  181	        isError=is_error,
  182	    )

  185	def _find_api_tool_error(exc: BaseException) -> ApiToolError | None:
  186	    seen: set[int] = set()
  187	    current: BaseException | None = exc
  188	    while current is not None and id(current) not in seen:
  189	        if isinstance(current, ApiToolError):
  190	            return current
  191	        seen.add(id(current))
  192	        current = current.__cause__ or current.__context__
  193	    return None
  194	
  195	
  196	def _root_exception(exc: BaseException) -> BaseException:
  197	    seen: set[int] = set()
  198	    current = exc
  199	    while id(current) not in seen:
  200	        seen.add(id(current))
  201	        next_exc = current.__cause__ or current.__context__
  202	        if next_exc is None:
  203	            break
  204	        current = next_exc
  205	    return current

  208	def _result_from_content(content: list[Any]) -> Any:
  209	    if len(content) != 1 or not isinstance(content[0], TextContent):
  210	        return None
  211	    text = content[0].text
  212	    try:
  213	        return json.loads(text)
  214	    except (json.JSONDecodeError, TypeError):
  215	        return text

  218	class OrchestraMCP(FastMCP):
  219	    async def call_tool(self, name: str, arguments: dict[str, Any]):
  220	        try:
  221	            if name in DISABLED_TOOLS:
  222	                raise ApiToolError(
  223	                    code="tool_disabled",
  224	                    message=f"Orchestra tool {name!r} is disabled for worker {WORKER_NAME!r} "
  225	                            f"(role {ROLE!r}) by role/worker disabled_tools policy. "
  226	                            "Ask the owner to change the policy and reconnect the worker.",
  227	                    details={"tool": name, "worker": WORKER_NAME, "role": ROLE},
  228	                )
  229	            converted = await super().call_tool(name, arguments)
  230	        except Exception as exc:
  231	            error = _find_api_tool_error(exc)
  232	            if error is None:
  233	                root = _root_exception(exc)
  234	                error = ApiToolError(
  235	                    code="tool_error",
  236	                    message=err_text(root),
  237	                    details={"exception_type": type(root).__name__, "tool": name},
  238	                )
  239	            envelope = _canonical_error(error)
  240	            logger.warning(
  241	                "MCP tool %s failed: code=%s request_id=%s message=%s",
  242	                name,
  243	                envelope["code"],
  244	                envelope["request_id"],
  245	                envelope["message"],
  246	            )
  247	            return mcp_tool_result(
  248	                result=error.result,
  249	                error=envelope,
  250	                is_error=True,
  251	            )
  252	
  253	        if isinstance(converted, CallToolResult):
  254	            structured = converted.structuredContent
  255	            if isinstance(structured, dict) and "result" in structured and "error" in structured:
  256	                raw_error = structured["error"]
  257	                if raw_error is None and converted.isError:
  258	                    text = "\n".join(
  259	                        item.text for item in converted.content if isinstance(item, TextContent)
  260	                    )
  261	                    raw_error = ApiToolError(code="tool_error", message=text or "MCP tool failed")
  262	                envelope = _canonical_error(raw_error) if raw_error is not None else None
  263	                content = list(converted.content)
  264	                if envelope:
  265	                    content = [
  266	                        item.model_copy(update={"text": _safe_response_text(item.text)})
  267	                        if isinstance(item, TextContent)
  268	                        else item
  269	                        for item in content
  270	                    ]
  271	                return CallToolResult(
  272	                    content=content,
  273	                    structuredContent={"result": structured["result"], "error": envelope},
  274	                    isError=converted.isError,
  275	                )
  276	            result = structured if structured is not None else _result_from_content(list(converted.content))
  277	            if converted.isError:
  278	                text = "\n".join(
  279	                    item.text for item in converted.content if isinstance(item, TextContent)
  280	                )
  281	                error = ApiToolError(code="tool_error", message=text or "MCP tool failed")
  282	                return mcp_tool_result(result=result, error=error, is_error=True, text=text)
  283	            return CallToolResult(
  284	                content=list(converted.content),
  285	                structuredContent={"result": result, "error": None},
  286	                isError=False,
  287	            )
  288	
  289	        if isinstance(converted, tuple):
  290	            content, structured = converted
  291	            tool = self._tool_manager.get_tool(name)
  292	            if (
  293	                tool is not None
  294	                and tool.fn_metadata.wrap_output
  295	                and isinstance(structured, dict)
  296	                and set(structured) == {"result"}
  297	            ):
  298	                result = structured["result"]
  299	            else:
  300	                result = structured
  301	            return CallToolResult(
  302	                content=list(content),
  303	                structuredContent={"result": result, "error": None},
  304	                isError=False,
  305	            )
  306	
  307	        content = list(converted)
  308	        return CallToolResult(
  309	            content=content,
  310	            structuredContent={"result": _result_from_content(content), "error": None},
  311	            isError=False,
  312	        )

  318	def _tool_names_for_access_mode(names: set[str], mode: str) -> set[str]:
  319	    normalized = mode.strip().lower()
  320	    if normalized in {"read-only", "readonly", "read"}:
  321	        return names & READ_ONLY_MCP_TOOLS
  322	    if normalized == "reducer":
  323	        return names & REDUCER_MCP_TOOLS
  324	    if normalized == "full":
  325	        return set(names)
  326	    raise ValueError(f"Unknown ORCHESTRA_ACCESS_MODE: {mode!r}")

  329	def _apply_access_mode() -> None:
  330	    registered = {tool.name for tool in mcp._tool_manager.list_tools()}
  331	    visible = _tool_names_for_access_mode(registered, ACCESS_MODE)
  332	    for name in registered - visible:
  333	        mcp.remove_tool(name)
  334	    logger.info(
  335	        "Orchestra MCP access=%s tools=%d/%d",
  336	        ACCESS_MODE,
  337	        len(visible),
  338	        len(registered),
  339	    )

  353	def _retry_after_seconds(value: Any) -> int | float | None:
  354	    if value is None:
  355	        return None
  356	    try:
  357	        parsed = float(value)
  358	    except (TypeError, ValueError):
  359	        return None
  360	    if not math.isfinite(parsed) or parsed < 0:
  361	        return None
  362	    return int(parsed) if parsed.is_integer() else parsed

  365	_SENSITIVE_DETAIL_KEYS = ("authorization", "token", "password", "secret", "api_key", "api-key")
  366	
  367	
  368	def _safe_detail(value: Any, depth: int = 0) -> Any:
  369	    if depth > 4:
  370	        return "[truncated]"
  371	    if isinstance(value, dict):
  372	        return {
  373	            str(key): (
  374	                "[redacted]"
  375	                if any(marker in str(key).lower() for marker in _SENSITIVE_DETAIL_KEYS)
  376	                else _safe_detail(item, depth + 1)
  377	            )
  378	            for key, item in list(value.items())[:50]
  379	        }
  380	    if isinstance(value, list):
  381	        return [_safe_detail(item, depth + 1) for item in value[:50]]
  382	    if isinstance(value, str):
  383	        return _safe_response_text(value)
  384	    return value
  385	
  386	
  387	def _safe_response_text(value: str) -> str:
  388	    text = re.sub(
  389	        r"(?i)\b(Bearer|Basic)\s+[^\s,;}\]]+",
  390	        r"\1 [redacted]",
  391	        value[:4000],
  392	    )
  393	    text = re.sub(
  394	        r"(?i)((?:authorization|token|password|secret|api[_-]?key)[\"']?\s*[:=]\s*)"
  395	        r"(?:\"[^\"\r\n]*\"|'[^'\r\n]*'|[^,;}&\r\n]+)",
  396	        r"\1[redacted]",
  397	        text,
  398	    )
  399	    return text[:1000]

  402	def _response_error(
  403	    method: str,
  404	    path: str,
  405	    response: httpx.Response,
  406	    payload: Any,
  407	    request_id: str,
  408	) -> ApiToolError:
  409	    body = payload if isinstance(payload, dict) else {}
  410	    raw_error = body.get("error")
  411	    server_error = raw_error if isinstance(raw_error, dict) else {}
  412	
  413	    message_value: Any = None
  414	    if isinstance(raw_error, str):
  415	        message_value = raw_error
  416	    elif server_error:
  417	        message_value = server_error.get("message") or server_error.get("detail")
  418	    if message_value is None:
  419	        message_value = body.get("message") or body.get("detail")
  420	    if message_value is None and not isinstance(payload, dict):
  421	        message_value = _safe_response_text(response.text)
  422	    if isinstance(message_value, str):
  423	        message = _safe_response_text(message_value.strip())
  424	    elif message_value is not None:
  425	        message = json.dumps(message_value, ensure_ascii=False)[:1000]
  426	    else:
  427	        message = ""
  428	    if not message:
  429	        message = f"HTTP {response.status_code}" if response.status_code >= 400 else "API returned an error"
  430	
  431	    if server_error.get("code"):
  432	        code = str(server_error["code"])
  433	    elif body.get("code"):
  434	        code = str(body["code"])
  435	    elif response.status_code == 429:
  436	        code = "http_429"
  437	    elif response.status_code >= 500:
  438	        code = "http_5xx"
  439	    elif response.status_code >= 400:
  440	        code = "http_4xx"
  441	    else:
  442	        code = "domain_error"
  443	
  444	    response_request_id = (
  445	        server_error.get("request_id")
  446	        or body.get("request_id")
  447	        or response.headers.get("X-Request-ID")
  448	        or response.headers.get("X-Correlation-ID")
  449	        or request_id
  450	    )
  451	    retry_after = _retry_after_seconds(
  452	        server_error.get("retry_after_seconds")
  453	        or body.get("retry_after_seconds")
  454	        or response.headers.get("Retry-After")
  455	    )
  456	    transient_status = response.status_code in {408, 425, 429} or response.status_code >= 500
  457	    outcome_unknown = method != "GET" and response.status_code >= 500
  458	    if isinstance(server_error.get("outcome_unknown"), bool):
  459	        outcome_unknown = server_error["outcome_unknown"]
  460	    retryable = method == "GET" and transient_status
  461	    if isinstance(server_error.get("retryable"), bool):
  462	        retryable = server_error["retryable"]
  463	    if outcome_unknown:
  464	        retryable = False
  465	
  466	    details: dict[str, Any] = {"method": method, "path": path}
  467	    server_details = server_error.get("details")
  468	    if isinstance(server_details, dict):
  469	        details["server"] = _safe_detail(server_details)
  470	    elif response.text:
  471	        details["response_body"] = _safe_response_text(response.text)
  472	    return ApiToolError(
  473	        code=code,
  474	        message=message,
  475	        status=response.status_code,
  476	        retryable=retryable,
  477	        request_id=str(response_request_id),
  478	        retry_after_seconds=retry_after,
  479	        outcome_unknown=outcome_unknown,
  480	        details=details,
  481	        result=body.get("result"),
  482	    )

  485	def _transport_error(
  486	    method: str,
  487	    path: str,
  488	    exc: httpx.RequestError,
  489	    request_id: str,
  490	) -> ApiToolError:
  491	    connect_failure = isinstance(exc, (httpx.ConnectError, httpx.ConnectTimeout))
  492	    no_request_sent = connect_failure or isinstance(exc, httpx.PoolTimeout)
  493	    timeout = isinstance(exc, httpx.TimeoutException)
  494	    return ApiToolError(
  495	        code="connect_error" if connect_failure else ("transport_timeout" if timeout else "transport_error"),
  496	        message=err_text(exc),
  497	        retryable=method == "GET" or no_request_sent,
  498	        request_id=request_id,
  499	        outcome_unknown=method != "GET" and not no_request_sent,
  500	        details={
  501	            "method": method,
  502	            "path": path,
  503	            "exception_type": type(exc).__name__,
  504	            "request_not_sent": no_request_sent,
  505	        },
  506	    )

  509	async def _api(method: str, path: str, **kwargs) -> dict | list | None:
  510	    # New client per call: avoids shared state across tool invocations in the same MCP session
  511	    method = method.upper()
  512	    request_id = str(kwargs.pop("request_id", "") or uuid.uuid4().hex)
  513	    idempotency_key = str(kwargs.pop("idempotency_key", "") or "")
  514	    if method not in {"GET", "POST", "PUT", "DELETE"}:
  515	        raise ApiToolError(
  516	            code="unsupported_method",
  517	            message=f"Unsupported HTTP method: {method}",
  518	            request_id=request_id,
  519	            details={"method": method, "path": path},
  520	        )
  521	    t = kwargs.pop("timeout", 30)
  522	    headers = _auth_headers()
  523	    headers["X-Request-ID"] = request_id
  524	    if idempotency_key:
  525	        headers["Idempotency-Key"] = idempotency_key
  526	    started = time.monotonic()
  527	    try:
  528	        async with httpx.AsyncClient(base_url=ORCHESTRA_URL, timeout=t, headers=headers) as client:
  529	            if method == "GET":
  530	                response = await client.get(path, params=kwargs.get("params"))
  531	            elif method == "POST":
  532	                response = await client.post(path, json=kwargs.get("json"))
  533	            elif method == "PUT":
  534	                response = await client.put(path, json=kwargs.get("json"), params=kwargs.get("params"))
  535	            else:
  536	                response = await client.delete(path, params=kwargs.get("params"))
  537	    except httpx.RequestError as exc:
  538	        error = _transport_error(method, path, exc, request_id)
  539	        error.details.update({
  540	            "elapsed_seconds": round(time.monotonic() - started, 3),
  541	            "timeout_seconds": httpx.Timeout(t).as_dict(),
  542	        })
  543	        raise error from exc
  544	
  545	    try:
  546	        payload = response.json()
  547	    except (json.JSONDecodeError, ValueError) as exc:
  548	        if response.status_code >= 400:
  549	            raise _response_error(method, path, response, None, request_id) from exc
  550	        raise ApiToolError(
  551	            code="invalid_response",
  552	            message=f"Invalid JSON response (status={response.status_code})",
  553	            status=response.status_code,
  554	            retryable=False,
  555	            request_id=(response.headers.get("X-Request-ID") or request_id),
  556	            outcome_unknown=method != "GET",
  557	            details={
  558	                "method": method,
  559	                "path": path,
  560	                "response_body": _safe_response_text(response.text),
  561	                "exception_type": type(exc).__name__,
  562	            },
  563	        ) from exc
  564	
  565	    if response.status_code >= 400:
  566	        raise _response_error(method, path, response, payload, request_id)
  567	    if isinstance(payload, dict) and payload.get("error") is not None:
  568	        raise _response_error(method, path, response, payload, request_id)
  569	    return payload

  572	def _spawn_delivery_error(
  573	    name: str,
  574	    mapping: dict[str, str],
  575	    cause: ApiToolError,
  576	    *,
  577	    task: str,
  578	    delivery_id: str,
  579	) -> ApiToolError:
  580	    if cause.code == "IDEMPOTENCY_CONFLICT" or cause.status == 409:
  581	        next_action = {
  582	            "code": "RESOLVE_IDEMPOTENCY_CONFLICT",
  583	            "delivery_id": delivery_id,
  584	            "message": (
  585	                "This delivery_id belongs to another payload. Repeat only the original "
  586	                "spawn_worker call with the same task and delivery_id; do not reuse this "
  587	                "id for changed input."
  588	            ),
  589	        }
  590	        delivery = "failed"
  591	    elif cause.outcome_unknown:
  592	        delivery = "unknown"
  593	        next_action = {
  594	            "code": "REPEAT_SPAWN_SAME_ID",
  595	            "delivery_id": delivery_id,
  596	            "message": (
  597	                "Repeat the same spawn_worker call with the same name, task and "
  598	                f"delivery_id={delivery_id}; it returns the existing worker and "
  599	                "initial-task receipt without creating a duplicate."
  600	            ),
  601	        }
  602	    elif (
  603	        cause.code == "DELIVERY_ACCEPT_REJECTED" and cause.status == 503
  604	    ) or cause.details.get("request_not_sent") is True or cause.code == "connect_error":
  605	        delivery = "failed"
  606	        next_action = {
  607	            "code": "RETRY_SAME_DELIVERY",
  608	            "tool": "retry_initial_delivery",
  609	            "arguments": {"name": name, "task": task, "delivery_id": delivery_id},
  610	            "message": "Retry only this delivery id; do not create a new logical task.",
  611	        }
  612	    else:
  613	        delivery = "failed"
  614	        next_action = {
  615	            "code": "RESOLVE_DELIVERY_REJECTION",
  616	            "message": "Resolve this typed delivery rejection; do not resend automatically.",
  617	        }
  618	    result = {
  619	        "worker_name": name,
  620	        "created": True,
  621	        "delivery": delivery,
  622	        "delivery_id": delivery_id,
  623	        **mapping,
  624	        "next_action": next_action,
  625	    }
  626	    details = dict(cause.details)
  627	    details.update({
  628	        "phase": "initial_task_delivery",
  629	        "delivery_id": delivery_id,
  630	        "next_action": next_action,
  631	    })
  632	    return ApiToolError(
  633	        code=cause.code,
  634	        message=(
  635	            f"Worker '{name}' was created, but initial task delivery {delivery}: "
  636	            f"{cause.message}; delivery_id={delivery_id}"
  637	        ),
  638	        status=cause.status,
  639	        retryable=False,
  640	        request_id=cause.request_id,
  641	        retry_after_seconds=cause.retry_after_seconds,
  642	        outcome_unknown=cause.outcome_unknown,
  643	        details=details,
  644	        result=result,
  645	    )

  674	def _delivery_receipt_text(
  675	    name: str,
  676	    model: str,
  677	    mapping: dict[str, str],
  678	    delivery: dict[str, Any],
  679	) -> str:
  680	    delivery_id = str(delivery.get("delivery_id") or "?")
  681	    state = str(delivery.get("delivery_state") or delivery.get("state") or "UNKNOWN")
  682	    # Сверка после сбоя транспорта возвращает запись в ЛЮБОМ состоянии, включая
  683	    # терминальный отказ до отправки. «Task accepted» на нём — молчаливая потеря:
  684	    # воркер создан и стоит без задания, а спавнящий читает приёмку и уходит.
  685	    if state == "FAILED_BEFORE_SUBMIT":
  686	        headline = (
  687	            "Task was NOT delivered — the worker has no task yet. Resend the SAME task "
  688	            f"text with retry_initial_delivery(name='{name}', delivery_id='{delivery_id}')."
  689	        )
  690	    elif state == "WAITING_QUOTA":
  691	        action = delivery.get("next_action") if isinstance(delivery.get("next_action"), dict) else {}
  692	        error = delivery.get("error") if isinstance(delivery.get("error"), dict) else {}
  693	        details = error.get("details") if isinstance(error.get("details"), dict) else {}
  694	        reason = str(details.get("reason") or "quota gate is closed")
  695	        headline = (
  696	            "Task delivery is deferred by the quota gate (WAITING_QUOTA); it will go out "
  697	            f"automatically when the gate opens. Reason: {reason}. "
  698	            f"{_safe_response_text(str(action.get('message') or ''))}"
  699	        )
  700	    elif state in {"QUEUED", "PREPARING"}:
  701	        headline = "Task accepted; delivery is proceeding now and the quota gate is not holding it."
  702	    elif state == "SUBMITTED":
  703	        headline = "Task was submitted; the quota gate did not hold it."
  704	    elif state == "CANCELLED":
  705	        headline = (
  706	            "The initial task was cancelled before provider submission; this worker has no task. "
  707	            "Use send_message to assign it work or kill_worker to archive the worker."
  708	        )
  709	    elif state in {"DELIVERY_UNKNOWN", "UNKNOWN"}:
  710	        # Та же сверка может вернуть запись, чей исход у провайдера неизвестен, — или
  711	        # ответ вовсе без состояния (`_normalize_delivery_receipt` подставляет UNKNOWN).
  712	        # «Task accepted» здесь читается как доставка, которую никто не подтверждал.
  713	        headline = (
  714	            "Task delivery outcome is UNKNOWN — repeat the same spawn_worker call "
  715	            f"with delivery_id={delivery_id}; the call resumes this worker and "
  716	            "returns the same initial-task receipt without a duplicate."
  717	        )
  718	    else:
  719	        headline = "Task accepted."
  720	    out = (
  721	        f"Worker '{name}' spawned. Model: {model}. {headline} "
  722	        f"delivery_id={delivery_id}; state={state}."
  723	    )
  724	    out += (
  725	        f"\nWorktree: {mapping['worktree_path']}"
  726	        f"\nRepository: {mapping['repo_path']}"
  727	        f"\nGit common dir: {mapping['git_common_dir']}"
  728	        f"\nBranch: {mapping['branch']}"
  729	    )
  730	    return out

  734	    return (
  735	        isinstance(value, dict)
  736	        and value.get("delivery_id") == delivery_id
  737	        and bool(value.get("delivery_state") or value.get("state"))
  738	    )

  741	def _normalize_delivery_receipt(
  742	    value: Any,
  743	    delivery_id: str,
  744	    *,
  745	    require_state: bool = True,
  746	) -> dict[str, Any] | None:
  747	    if not isinstance(value, dict) or value.get("ok") is not True:
  748	        return None
  749	    response_delivery_id = value.get("delivery_id")
  750	    if response_delivery_id is not None and response_delivery_id != delivery_id:
  751	        return None
  752	    delivery_state = value.get("delivery_state") or value.get("state")
  753	    if require_state and not delivery_state:
  754	        return None
  755	    return {
  756	        **value,
  757	        "delivery_id": response_delivery_id or delivery_id,
  758	        "delivery_state": delivery_state or "UNKNOWN",
  759	    }

  762	async def _reconcile_initial_delivery(
  763	    cause: ApiToolError,
  764	    delivery_id: str,
  765	    scope: str,
  766	    *,
  767	    allow_legacy_status: bool = False,
  768	) -> tuple[dict[str, Any] | None, ApiToolError]:
  769	    is_conflict = cause.code == "IDEMPOTENCY_CONFLICT" or cause.status == 409
  770	    should_reconcile = (
  771	        not is_conflict and cause.outcome_unknown
  772	        or (cause.status is not None and cause.status >= 500
  773	            and not is_conflict
  774	            and not (cause.code == "DELIVERY_ACCEPT_REJECTED"
  775	                     and cause.status == 503
  776	                     and not cause.outcome_unknown))
  777	    )
  778	    if not should_reconcile:
  779	        return None, cause
  780	    try:
  781	        status = await _api(
  782	            "GET", _delivery_status_path(delivery_id), params={"scope": scope},
  783	        )
  784	    except ApiToolError as status_error:
  785	        reconciliation: Any = status_error.envelope()
  786	    else:
  787	        if _is_delivery_receipt(status, delivery_id):
  788	            return status, cause
  789	        if allow_legacy_status:
  790	            status_receipt = _normalize_delivery_receipt(
  791	                status, delivery_id, require_state=False,
  792	            )
  793	            if status_receipt is not None:
  794	                return status_receipt, cause
  795	        reconciliation = {"status": "missing", "response": status}
  796	    details = dict(cause.details)
  797	    details["delivery_id"] = delivery_id
  798	    details["reconciliation"] = reconciliation
  799	    return None, ApiToolError(
  800	        code=cause.code,
  801	        message=cause.message,
  802	        status=cause.status,
  803	        retryable=False,
  804	        request_id=cause.request_id,
  805	        retry_after_seconds=cause.retry_after_seconds,
  806	        outcome_unknown=True,
  807	        details=details,
  808	    )

  812	    name: str,
  813	    task: str,
  814	    delivery_id: str,
  815	    scope: str = SCOPE,
  816	) -> dict[str, Any]:

  818	    try:
  819	        result = await _api(

  822	    except ApiToolError as cause:
  823	        status, cause = await _reconcile_initial_delivery(cause, delivery_id, scope)
  824	        if status is not None:

  826	        raise cause
  827	    if isinstance(result, dict) and result.get("error") is not None:
  828	        raw_error = result["error"]

  832	                message=str(raw_error.get("message") or raw_error.get("detail") or "delivery rejected"),
  833	                status=raw_error.get("status") if isinstance(raw_error.get("status"), int) else 200,
  834	                retryable=bool(raw_error.get("retryable", False)),
  835	                request_id=raw_error.get("request_id"),
  836	                outcome_unknown=bool(raw_error.get("outcome_unknown", False)),
  837	                details=raw_error.get("details") if isinstance(raw_error.get("details"), dict) else {},
  838	            )
  839	        else:
  840	            cause = ApiToolError(
  841	                code="domain_error",
  842	                message=str(raw_error),
  843	                status=200,
  844	                retryable=False,
  845	                outcome_unknown=False,
  846	                details={"response": result},
  847	            )
  848	        status, cause = await _reconcile_initial_delivery(cause, delivery_id, scope)
  849	        if status is not None:
  850	            return status
  851	        raise cause
  852	    normalized = _normalize_delivery_receipt(result, delivery_id)
  853	    if normalized is None:
  854	        cause = ApiToolError(
  855	            code="invalid_response",
  856	            message="Initial delivery API returned an invalid acceptance receipt",
  857	            status=200,
  858	            outcome_unknown=True,
  859	            details={"phase": "initial_task_delivery", "response": result},
  860	        )
  861	        status, cause = await _reconcile_initial_delivery(
  862	            cause, delivery_id, scope, allow_legacy_status=True,
  863	        )
  864	        if status is not None:
  865	            return status
  866	        raise cause
  867	    return normalized

  971	def _cross_repo_note(scope: str, mapping: dict) -> str:
  972	    """Предупреждение, когда воркер заведён в РЕПОЗИТОРИИ, отличном от проекта родителя.
  973	
  974	    Сравнение идёт по git common dir, а не по строкам путей: worktree всегда лежит внутри

  979	        return ""
  980	    common = str(mapping.get("git_common_dir") or "")
  981	    if not common:
  982	        return ""
  983	    scope_git = os.path.join(scope.rstrip("/"), ".git")
  984	    try:
  985	        same = os.path.realpath(common) == os.path.realpath(scope_git)
  986	    except OSError:
  987	        same = common.rstrip("/") == scope_git
  988	    if same:
  989	        return ""
  990	    return (
  991	        f"⚠️ ДРУГОЙ РЕПОЗИТОРИЙ: воркер коммитит в {mapping.get('repo_path')}, "

  993	        "проектов, но помни: задачи и их номера остаются в проекте родителя, а "

  999	async def _resume_cut_off_spawn(
 1000	    name: str, task: str, scope: str, delivery_id: str, cause: ApiToolError,
 1001	) -> dict[str, Any]:
 1002	    """Create call was cut off (V-695): the server may still have made the worker.
 1003	
 1004	    Timeout/transport loss or "already exists" → ask the server whether this very spawn
 1005	    (same delivery_id and task) owns the worker; if so, continue to the idempotent delivery.
 1006	    """
 1007	    cut_off = cause.outcome_unknown and cause.status is None
 1008	    taken = cause.status == 409 and "already exists" in cause.message
 1009	    if not (cut_off or taken):
 1010	        raise cause
 1011	    try:
 1012	        return await _api(
 1013	            "POST", f"/api/sessions/{name}/spawn-resume",
 1014	            json={"scope": scope, "delivery_id": delivery_id, "task": task},
 1015	        )
 1016	    except ApiToolError as verdict:
 1017	        if verdict.status == 404 and cut_off:
 1018	            cause.details["next_action"] = {
 1019	                "code": "RETRY_SPAWN_SAME_DELIVERY_ID",
 1020	                "message": (
 1021	                    "Creation outcome unknown. Repeat spawn_worker with the same name, task "
 1022	                    f"and delivery_id={delivery_id}: it resumes the worker if it was created "
 1023	                    "and never delivers the task twice."
 1024	                ),
 1025	                "delivery_id": delivery_id,
 1026	            }
 1027	            cause.details["delivery_id"] = delivery_id
 1028	            cause.message = (
 1029	                f"{cause.message}; delivery_id={delivery_id}. "
 1030	                f"{cause.details['next_action']['message']}"
 1031	            )
 1032	            cause.result = {
 1033	                "worker_name": name,
 1034	                "created": "unknown",
 1035	                "delivery_id": delivery_id,
 1036	                "next_action": cause.details["next_action"],
 1037	            }
 1038	            raise cause from verdict
 1039	        next_action = {
 1040	            "code": "REPEAT_SPAWN_SAME_ID",
 1041	            "delivery_id": delivery_id,
 1042	            "message": (
 1043	                "The worker creation outcome could not be reconciled. Repeat the "
 1044	                "same spawn_worker call with the same name, task and "
 1045	                f"delivery_id={delivery_id}; "
 1046	                "it resumes an existing worker rather than creating a duplicate."
 1047	            ),
 1048	        }
 1049	        verdict.details = {
 1050	            **verdict.details,
 1051	            "delivery_id": delivery_id,
 1052	            "next_action": next_action,
 1053	        }
 1054	        verdict.result = {
 1055	            "worker_name": name,
 1056	            "created": "unknown",
 1057	            "delivery_id": delivery_id,
 1058	            "next_action": next_action,
 1059	        }
 1060	        verdict.message = f"{verdict.message}; {next_action['message']}"
 1061	        raise verdict from cause

 1407	def _message_delivery_receipt_text(
 1408	    receipt: dict[str, Any], *, to: str = "", parent_name: str = "",
 1409	) -> str:
 1410	    delivery_id = str(receipt["delivery_id"])
 1411	    state = str(receipt.get("delivery_state") or receipt.get("state") or "UNKNOWN")
 1412	    target = f" to '{to}'" if to else ""
 1413	    error = receipt.get("error") if isinstance(receipt.get("error"), dict) else {}
 1414	    code = str(error.get("code") or "")
 1415	    message = _safe_response_text(str(error.get("message") or ""))
 1416	    action = receipt.get("next_action") if isinstance(receipt.get("next_action"), dict) else {}
 1417	    if action.get("code") == "TARGET_QUEUE_BLOCKED":
 1418	        # Условие блокировки принадлежит message_deliveries._queue_block; здесь только
 1419	        # рендер его вывода. Свой пересчёт «голова очереди неразобрана» разошёлся бы с
 1420	        # владельцем, а отправитель за барьером читал бы бодрый QUEUED — как те 25 часов.
 1421	        output = (
 1422	            f"Message NOT delivered{target}; delivery_id={delivery_id}; state={state}.\n"
 1423	            f"{_safe_response_text(str(action.get('message') or ''))}"
 1424	        )
 1425	    elif state == "WAITING_QUOTA" or action.get("code") == "BEHIND_QUOTA_WAIT":
 1426	        # Не отказ и не «доставлено»: платформа сама держит сообщение за гейтом квот.
 1427	        output = (
 1428	            f"Message delivery is deferred by a quota gate{target}; "
 1429	            f"delivery_id={delivery_id}; state={state}.\n"
 1430	            f"{_safe_response_text(str(action.get('message') or ''))}\n"
 1431	            f"Withdraw it with cancel_message_delivery(delivery_id=\"{delivery_id}\")."
 1432	        )
 1433	    elif state == "CANCELLED":
 1434	        output = f"Message cancelled{target}; delivery_id={delivery_id}; it was not sent."
 1435	    elif state == "WAITING_NEXT_TURN":
 1436	        output = (
 1437	            f"Message is waiting for the target's next turn{target}; "
 1438	            f"delivery_id={delivery_id}; state={state}.\n"
 1439	            f"{_safe_response_text(str(action.get('message') or ''))}"
 1440	        )
 1441	    elif state == "DELIVERY_UNKNOWN":
 1442	        output = (
 1443	            f"Message delivery outcome is unknown{target}; delivery_id={delivery_id}; "
 1444	            f"code={code or 'DELIVERY_OUTCOME_UNKNOWN'}: "
 1445	            f"{message or 'Provider outcome is unknown'}.\n"
 1446	            "Repeat this send_message call with the same target, text and "
 1447	            f"delivery_id={delivery_id} to receive the same receipt without a duplicate."
 1448	        )
 1449	    elif state == "FAILED_BEFORE_SUBMIT":
 1450	        # Терминальный отказ ДО отправки: провайдер сообщения не видел. Печатать его как
 1451	        # приёмку — молчаливая потеря задания (штатный исход QuotaGateError), отправитель
 1452	        # читает acceptance и уходит дальше, а повтора не будет никогда.
 1453	        if error.get("retryable") is False:
 1454	            # Повтор тем же id перепроверяется против ЗАМОРОЖЕННОГО target_generation
 1455	            # (routes/sessions.py отдаёт его из существующей строки), поэтому у
 1456	            # TARGET_TASK_CHANGED причина по построению не исчезает: «повтори тем же
 1457	            # id» — вечный цикл, доставить может только новое сообщение.
 1458	            next_step = (
 1459	                "this delivery id is dead — retrying it repeats the same refusal. "
 1460	                "Send a new message with a new delivery_id to the target's current task."
 1461	            )
 1462	        else:
 1463	            next_step = (
 1464	                "once the cause is gone, resend the SAME text with the same "
 1465	                f"delivery_id=\"{delivery_id}\" — a new id would create a second "
 1466	                "logical message."
 1467	            )
 1468	        output = (
 1469	            f"Message was NOT delivered{target}; delivery_id={delivery_id}; "
 1470	            f"state={state}; code={code or 'DELIVERY_NOT_SUBMITTED'}: "
 1471	            f"{message or 'Provider submission did not begin'}.\n"
 1472	            f"Nothing reached the target: {next_step}"
 1473	        )
 1474	    elif state == "DELIVERY_UNKNOWN_ORPHANED":
 1475	        # Рестарт снял БЛОКИРОВКУ очереди, а не выяснил судьбу сообщения: процесс, который
 1476	        # мог его дослать, мёртв, но дошло оно или нет — так и неизвестно. «Accepted» здесь
 1477	        # читается как доставка, которой никто не подтверждал.

 1481	            "which did not deliver this message.\n"
 1482	            "Do not read this as delivered; confirm with the target before acting."
 1483	        )
 1484	    elif state == "QUEUED":
 1485	        output = (
 1486	            f"Message accepted; delivery is proceeding now{target}, and the quota gate is not holding it. "
 1487	            f"delivery_id={delivery_id}; state={state}."
 1488	        )
 1489	    elif state == "SUBMITTED":
 1490	        output = (
 1491	            f"Message was submitted{target}; the quota gate did not hold it. "
 1492	            f"delivery_id={delivery_id}; state={state}."
 1493	        )
 1494	    else:
 1495	        output = f"Message accepted{target}; delivery_id={delivery_id}; state={state}."

 1498	        "After an ambiguous call outcome, repeat this send_message call with the same delivery_id."
 1499	    )
 1500	    if parent_name and parent_name != WORKER_NAME:
 1501	        output += (
 1502	            f"\n⚠️ This worker belongs to '{parent_name}'. "

 1504	        )

 1621	def _cache_pill(s: dict) -> str:
 1622	    """Prompt-cache warmth as a short exact/approximate text pill."""
 1623	    from datetime import datetime, timezone
 1624	    from app.models import cache_policy_for_runtime, runtime_for_record
 1625	
 1626	    policy = cache_policy_for_runtime(runtime_for_record(s))
 1627	    raw_ttl = s.get("cache_ttl_seconds")
 1628	    raw_ttl = policy["cache_ttl_seconds"] if raw_ttl is None else raw_ttl
 1629	    try:
 1630	        ttl = int(raw_ttl)
 1631	    except (TypeError, ValueError):
 1632	        return ""
 1633	    if ttl <= 0:
 1634	        return ""
 1635	    approximate = bool(s.get("cache_ttl_approximate", policy["cache_ttl_approximate"]))
 1636	
 1637	    if s.get("status") in ("running", "starting"):
 1638	        return f"🔥 hot ≈{ttl // 60}m" if approximate else "🔥 hot"
 1639	    ts = s.get("last_turn_ts")
 1640	    if not ts:
 1641	        return ""
 1642	    try:
 1643	        last = datetime.fromisoformat(ts.replace("Z", "+00:00"))
 1644	    except (ValueError, AttributeError):
 1645	        return ""
 1646	    now = datetime.now(timezone.utc)
 1647	    elapsed_min = max(0, int((now.timestamp() - last.timestamp()) // 60))
 1648	    rem_min = (ttl // 60) - elapsed_min
 1649	    if rem_min <= 0:
 1650	        if approximate:
 1651	            past_reference = max(0, elapsed_min - ttl // 60)
 1652	            if past_reference < 60:
 1653	                past_label = f"{past_reference}m"
 1654	            elif past_reference < 1440:
 1655	                hours, minutes = divmod(past_reference, 60)
 1656	                past_label = f"{hours}h{f'{minutes}m' if minutes else ''}"
 1657	            else:
 1658	                days, hours = divmod(past_reference // 60, 24)
 1659	                past_label = f"{days}d{f'{hours}h' if hours else ''}"
 1660	            return f"🧊? unknown (+{past_label} past ≈{ttl // 60}m)"
 1661	        return "🧊 cold"
 1662	    marker = "≈" if approximate else ""
 1663	    ttl_min = ttl / 60
 1664	    if rem_min > ttl_min * 0.5:
 1665	        return f"🔥 hot {marker}{rem_min}m"
 1666	    if rem_min >= ttl_min * 0.2:
 1667	        return f"🟡 warm {marker}{rem_min}m"
 1668	    return f"🔴 cooling {marker}{rem_min}m"

 2515	async def _await_merge_terminal(
 2516	    operation_id: str, result: dict[str, Any], budget: float | None = None,
 2517	) -> dict[str, Any]:
 2518	    """Дождаться терминального состояния, чтобы один вызов давал один ответ.
 2519	
 2520	    Прежний бюджет был 2 с (0.0+0.5+1.5), а 9 операций из 31 (29%) шли дольше —
 2521	    вызывающий получал RUNNING на РАБОТАЮЩЕМ мерже и рапортовал провал.
 2522	    """
 2523	    # Значение по умолчанию берём в момент ВЫЗОВА: константа в сигнатуре связывается при
 2524	    # импорте, и тест, подменяющий потолок, молча получил бы старое число.
 2525	    budget = _MERGE_WAIT_SECONDS if budget is None else budget
 2526	    loop = asyncio.get_running_loop()
 2527	    deadline = loop.time() + budget
 2528	    delay = 0.2
 2529	    while True:
 2530	        remaining = deadline - loop.time()
 2531	        if remaining <= 0:
 2532	            return result
 2533	        try:
 2534	            async with asyncio.timeout(remaining):
 2535	                recovered = await _recover_merge_status(operation_id)
 2536	        except asyncio.TimeoutError:
 2537	            return result
 2538	        if recovered is not None:
 2539	            result = recovered
 2540	            if result.get("operation_state") not in {"PENDING", "RUNNING"}:
 2541	                return result
 2542	        remaining = deadline - loop.time()
 2543	        if remaining <= 0:
 2544	            return result
 2545	        await asyncio.sleep(min(delay, remaining))
 2546	        delay = min(delay * 1.5, 2.0)

 3702	def _parse_review_numstat(raw: bytes) -> tuple[int, int, int]:
 3703	    """Return changed lines, files, and binary files from ``git --numstat -z``."""
 3704	    if not raw:
 3705	        return 0, 0, 0
 3706	    if not raw.endswith(b"\0"):
 3707	        raise ValueError("numstat output is not NUL-terminated")
 3708	    fields = raw.split(b"\0")
 3709	    index = 0
 3710	    lines = 0
 3711	    files = 0
 3712	    binaries = 0
 3713	    while index < len(fields) - 1:
 3714	        header = fields[index]
 3715	        index += 1
 3716	        parts = header.split(b"\t", 2)
 3717	        if len(parts) != 3:
 3718	            raise ValueError("numstat record has no add/delete/path fields")
 3719	        added, deleted, path = parts
 3720	        if not path:
 3721	            if index + 1 >= len(fields) or not fields[index] or not fields[index + 1]:
 3722	                raise ValueError("numstat rename record has no source/destination path")
 3723	            index += 2
 3724	        if added == b"-" or deleted == b"-":
 3725	            if added != b"-" or deleted != b"-":
 3726	                raise ValueError("numstat binary marker is incomplete")
 3727	            binaries += 1
 3728	        else:
 3729	            if not added.isdigit() or not deleted.isdigit():
 3730	                raise ValueError("numstat line counts are not decimal integers")
 3731	            lines += int(added) + int(deleted)
 3732	        files += 1
 3733	    return lines, files, binaries

 3904	def _validate_project_context(parsed: dict[str, Any]) -> list[tuple[str, str]]:
 3905	    unknown = sorted(set(parsed) - _PROJECT_CONTEXT_KEYS)
 3906	    if unknown:
 3907	        raise ValueError(f"{unknown[0]}|unknown field")
 3908	    if "schema_version" not in parsed:
 3909	        raise ValueError("schema_version|required field is missing")
 3910	    if parsed["schema_version"] != 1:
 3911	        raise ValueError("schema_version|must be 1")
 3912	    values = []
 3913	    for key, label in _PROJECT_CONTEXT_FIELDS:
 3914	        if key not in parsed:
 3915	            raise ValueError(f"{key}|required field is missing")
 3916	        value = parsed[key]
 3917	        if not isinstance(value, str) or not value.strip():
 3918	            raise ValueError(f"{key}|must be a non-empty string")
 3919	        value = value.strip()
 3920	        if value.startswith(_PROJECT_CONTEXT_PLACEHOLDER_PREFIX):
 3921	            raise ValueError(f"{key}|placeholder must be replaced")
 3922	        if "\n" in value or "\r" in value:
 3923	            raise ValueError(f"{key}|must be a single-line string")
 3924	        values.append((label, value))
 3925	    return values
