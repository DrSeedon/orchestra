   65	def _is_safe_rel(p: str) -> bool:
   66	    """True если ``p`` — безопасный относительный путь (без абсолютного и '..').
   67	
   68	    Защита изоляции: слои промпта/шаблоны не должны выходить за .orchestra/pipelines/<name>/.
   69	    """
   70	    from pathlib import PurePosixPath
   71	    if not p or p.startswith("/"):
   72	        return False
   73	    return ".." not in PurePosixPath(p).parts

   76	def _is_safe_component(value: str) -> bool:
   77	    """True for one portable path component (pipeline/skill identifiers)."""
   78	    return bool(
   79	        value
   80	        and value not in {".", ".."}
   81	        and "/" not in value
   82	        and "\\" not in value
   83	        and "\x00" not in value
   84	    )

   87	def _validate_skill_names(value: AllOrList | None) -> AllOrList | None:
   88	    if value is None or value == "all":
   89	        return value
   90	    for name in value:
   91	        if not _is_safe_component(name):
   92	            raise ValueError(f"unsafe skill name '{name}'")
   93	    return value

  262	class RoleSpec(BaseModel):
  263	    """Сырая роль из манифеста. Опциональные поля (model/skills/...) = None →
  264	    наследуются из defaults на резолве. kind/label — обязательны для контракта.

  274	    # Модули — переиспользуемые блоки промпта (prompts/modules/{m}.md), инлайнятся
  275	    # в system_prompt после слоёв роли. Пусто → ничего не добавляется.
  276	    modules: list[str] = Field(default_factory=list)
  277	    # Переопределения defaults (None → наследуем):
  278	    model: str | None = None
  279	    effort: EffortSpec | None = None
  280	    skills: AllOrList | None = None
  281	    mcp_servers: AllOrList | None = None
  282	    base_branch_strategy: BranchStrategy | None = None
  283	    inherit_claude_md: bool | None = None
  284	    docs_scaffold: bool | None = None
  285	    # Роле-специфика:
  286	    docs_dir: DocsDir | None = None
  287	    tg: Tg | None = None
  288	    when: str | None = None
  289	    not_for: str | None = None
  290	    description: str | None = None
  291	
  292	    @field_validator("model")
  293	    @classmethod
  294	    def _model_known(cls, v: str | None) -> str | None:
  295	        if v is not None and not _model_is_known(v):
  296	            raise ValueError(f"unknown model '{v}'")
  297	        return v
  298	
  299	    @field_validator("effort")
  300	    @classmethod
  301	    def _effort_keys(cls, v: EffortSpec | None) -> EffortSpec | None:
  302	        """Карта эффорта: нормализовать ключи-модели; неизвестная СТУПЕНЬ — отказ.
  303	
  304	        Два разных исхода у двух разных ошибок, и разница не косметическая:
  305	
  306	        * **Неизвестная ступень → `ValueError`** (манифест не грузится). Пропуск такой
  307	          строки не «оставляет как было», а ТИХО МЕНЯЕТ МАРШРУТ: опечатка в ключе Sol
  308	          роняет его на `default`, то есть с `xhigh` на `high`, — и с перечитыванием на
  309	          границе хода это уезжает живым агентам само. Список ступеней замкнут и известен
  310	          на момент загрузки, гонки инициализации здесь нет, поэтому опечатка однозначна.
  311	          Цена принята осознанно: спавн новых агентов падает, пока не починят; живые при
  312	          этом работают — `AgentSession._apply_manifest_effort` на сбой резолва оставляет
  313	          текущее значение и НЕ пересобирает бэкенды.
  314	        * **Неизвестный ключ модели → warning, ключ сохраняем.** Реестр моделей
  315	          досоздаётся на старте (`models.fetch_models_from_proxy`), манифест мог быть
  316	          разобран раньше него — отказ или выброс потеряли бы валидную строку из-за
  317	          порядка инициализации.
  318	
  319	        Скаляр не трогаем вовсе — обратная совместимость.
  320	        """
  321	        if not isinstance(v, dict):
  322	            return v
  323	        from app.backend_codex import CODEX_REASONING_EFFORTS  # владелец набора ступеней
  324	        out: dict[str, str] = {}
  325	        for key, val in v.items():
  326	            if val not in CODEX_REASONING_EFFORTS:
  327	                raise ValueError(
  328	                    f"effort['{key}']: unknown level '{val}'; "
  329	                    f"known={sorted(CODEX_REASONING_EFFORTS)}")
  330	            # Порядок проверок: рантайм раньше модели, иначе ключи "codex"/"grok" (они же
  331	            # алиасы моделей gpt-6.1-sol и grok-4.6) читались бы как модель — и ключ
  332	            # "codex" не покрывал бы Luna. Имена рантаймов здесь ЗАРЕЗЕРВИРОВАНЫ: выбрать
  333	            # ими конкретную модель нельзя, для этого есть полный id.
  334	            if key == "default" or _is_runtime(key):
  335	                out[key] = val
  336	            elif _model_is_known(key):
  337	                out[_canon_model(key)] = val
  338	            else:
  339	                # Незнакомый ключ НЕ выбрасываем: реестр моделей досоздаётся на старте
  340	                # (`fetch_models_from_proxy`), и манифест мог быть разобран раньше него —
  341	                # выброс сделал бы валидную строку невосстановимо потерянной. Ключ просто
  342	                # не совпадёт ни с одной моделью, пока она не появится.
  343	                logger.warning(
  344	                    f"effort: key '{key}' — не известная модель и не рантайм; "
  345	                    f"сработает только если такая модель появится в реестре")
  346	                out[key] = val
  347	        return out
  348

  365	class PipelineConfig(BaseModel):
  366	    """Сырой манифест пайплайна (роли с None-полями, до наследования)."""
  367	    model_config = ConfigDict(extra="forbid")
  368	    name: str
  369	    description: str = ""
  370	    validation: ValidationMode = "fail-closed"
  371	    defaults: Defaults = Field(default_factory=Defaults)
  372	    roles: dict[str, RoleSpec]
  373	
  374	    @model_validator(mode="after")
  375	    def _check_graph(self) -> "PipelineConfig":
  376	        """can_spawn ⊆ roles (кроме '*'): ссылка на несуществующую роль → ошибка."""
  377	        known = set(self.roles)
  378	        for rname, r in self.roles.items():
  379	            for child in r.can_spawn:
  380	                if child != "*" and child not in known:
  381	                    raise ValueError(
  382	                        f"pipeline '{self.name}': role '{rname}' can_spawn references "
  383	                        f"unknown role '{child}'. known={sorted(known)}")
  384	        return self

  427	@lru_cache(maxsize=64)
  428	def _load_pipeline_cached(name: str, path: Path, stamp: tuple[int, int]) -> PipelineConfig:
  429	    """Разбор манифеста. Ключ кэша включает (mtime_ns, размер) — правка даёт промах.
  430	
  431	    Размер в ключе не избыточен: пара «та же секунда + тот же размер» уже давала
  432	    ложное попадание в кеше байткода CPython, и стоила полутора часов отладки (#214).
  433	    """
  434	    text = path.read_text()
  435	    after = path.stat()
  436	    if (after.st_mtime_ns, after.st_size) != stamp:
  437	        # Файл писали, пока мы читали. Проверки схемы тут не защищают: НЕДОПИСАННЫЙ YAML
  438	        # бывает синтаксически валидным, и роль молча получила бы неполную карту эффорта —
  439	        # а с #214 такая карта уезжает живым агентам на их следующем ходе.
  440	        # lru_cache исключения не кеширует, поэтому отравленной записи не остаётся.
  441	        raise _ManifestUnstable(f"pipeline '{name}' changed while being read")
  442	    data = yaml.safe_load(text) or {}
  443	    cfg = PipelineConfig(**data)  # pydantic: схема + граф can_spawn
  444	    if cfg.name != name:
  445	        raise ValueError(f"pipeline name '{cfg.name}' != dir '{name}'")
  446	    return cfg

  449	def load_pipeline(name: str) -> PipelineConfig:
  450	    """Прочитать ``.orchestra/pipelines/<name>/pipeline.yaml``, провалидировать, кэшировать.
  451	
  452	    Кэш инвалидируется по (mtime_ns, размер): правка манифеста при живом сервере видна
  453	    сразу, без рестарта (#214 — эффорт перечитывается на границе каждого хода).
  454	    Один ``stat`` на вызов, разбор — только на изменившемся файле.
  455	
  456	    Чтение стабильное: если файл сменился ПОКА мы его читали, разбор отбрасывается и
  457	    попытка повторяется (до трёх раз). Неатомарная запись извне — редактором, `sed -i`,
  458	    деплоем — иначе отдала бы недописанный YAML, который бывает синтаксически валидным.
  459	
  460	    :raises FileNotFoundError: если папки/файла нет.
  461	    :raises pydantic.ValidationError: если схема битая (extra-поле, неверный kind/model).
  462	    :raises ValueError: если ``name`` в файле не совпадает с именем папки, либо битый
  463	        граф can_spawn.
  464	    :raises OSError: если файл менялся в течение всех трёх попыток чтения.
  465	    """
  466	    # Empty name = legacy/ad-hoc session with no pipeline → treat as "not found"
  467	    # so callers' FileNotFoundError branch handles it (role=None), not a crash.
  468	    if not name:
  469	        raise FileNotFoundError("pipeline name is empty")
  470	    if not _is_safe_component(name):
  471	        raise ValueError(f"unsafe pipeline name '{name}'")
  472	    path = PIPELINES_DIR / name / "pipeline.yaml"
  473	    if not path.is_file():
  474	        raise FileNotFoundError(f"pipeline '{name}' not found at {path}")
  475	    unstable: _ManifestUnstable | None = None
  476	    for _ in range(3):
  477	        st = path.stat()
  478	        try:
  479	            return _load_pipeline_cached(name, path, (st.st_mtime_ns, st.st_size))
  480	        except _ManifestUnstable as e:
  481	            unstable = e
  482	    raise unstable
  483	
  484	
  485	# Вызывающие (и ~23 теста) чистят кэш через load_pipeline.cache_clear() — адрес
  486	# сохраняем, чтобы разбиение на «stat + разбор» не расползлось по коду.
  487	load_pipeline.cache_clear = _load_pipeline_cached.cache_clear
  488	load_pipeline.cache_info = _load_pipeline_cached.cache_info

  503	def list_pipelines() -> list[dict]:
  504	    """Скан ``.orchestra/pipelines/`` (включая gitignored). Для UI-дропдауна.
  505	
  506	    Возвращает ``[{name, description, roles:[...], valid:bool, error:str|None}]``.
  507	    Битый манифест НЕ роняет список — помечается ``valid=False`` с текстом ошибки.
  508	    """
  509	    out: list[dict] = []
  510	    if not PIPELINES_DIR.is_dir():
  511	        return out
  512	    for d in sorted(PIPELINES_DIR.iterdir()):
  513	        if not d.is_dir() or not (d / "pipeline.yaml").is_file():
  514	            continue
  515	        try:
  516	            cfg = load_pipeline(d.name)
  517	            out.append({"name": cfg.name, "description": cfg.description,
  518	                        "roles": sorted(cfg.roles), "valid": True, "error": None})
  519	        except Exception as e:  # noqa: BLE001 — намеренно глотаем, чтобы список не падал
  520	            out.append({"name": d.name, "description": "", "roles": [],
  521	                        "valid": False, "error": str(e)})
  522	    return out

  527	def _merge_scalar(default_val, role_val):
  528	    """Скаляр: роль переопределяет, если задала (не None); иначе наследуем."""
  529	    return default_val if role_val is None else role_val

  532	def _merge_list(default_val: AllOrList, role_val: AllOrList | None) -> AllOrList:
  533	    """Список: union(defaults ∪ role). ``"all"`` в любом из двух → ``"all"`` (поглощает)."""
  534	    if role_val is None:
  535	        return default_val
  536	    if default_val == "all" or role_val == "all":
  537	        return "all"
  538	    return sorted(set(default_val) | set(role_val))

  541	def resolve_role(pipeline: PipelineConfig, role: str) -> ResolvedRole:
  542	    """Слить роль с defaults в :class:`ResolvedRole` (все поля заполнены).
  543	
  544	    Скаляр — роль переопределяет если задан, иначе defaults. Список (skills/
  545	    mcp_servers) — union с поглощением ``"all"``. ``prompt_layers`` — по kind роли
  546	    с подстановкой ``{role}``.
  547	
  548	    :raises KeyError: если ``role`` нет в ``pipeline.roles`` (ловит вызывающий).
  549	    """
  550	    spec = pipeline.roles[role]
  551	    d = pipeline.defaults
  552	    layers_tmpl = (d.prompt_layers.orchestrator if spec.kind == "orchestrator"
  553	                   else d.prompt_layers.worker)
  554	    return ResolvedRole(
  555	        name=role, pipeline=pipeline.name, kind=spec.kind, label=spec.label,
  556	        order=spec.order, can_spawn=spec.can_spawn,
  557	        allow_unrouted_workers=spec.allow_unrouted_workers,
  558	        modules=spec.modules, disabled_tools=spec.disabled_tools,
  559	        model=_merge_scalar(d.model, spec.model),
  560	        effort=spec.effort,
  561	        skills=_merge_list(d.skills, spec.skills),
  562	        mcp_servers=_merge_list(d.mcp_servers, spec.mcp_servers),
  563	        base_branch_strategy=_merge_scalar(d.base_branch_strategy, spec.base_branch_strategy),
  564	        inherit_claude_md=_merge_scalar(d.inherit_claude_md, spec.inherit_claude_md),
  565	        docs_scaffold=_merge_scalar(d.docs_scaffold, spec.docs_scaffold),
  566	        docs_dir=spec.docs_dir, tg=spec.tg,
  567	        when=spec.when, not_for=spec.not_for, description=spec.description,
  568	        prompt_layers=[p.replace("{role}", role) for p in layers_tmpl],
  569	    )

  572	def resolve_effort(effort: EffortSpec | None, model: str, runtime: str) -> str | None:
  573	    """Ступень эффорта под конкретную модель спавна.
  574	
  575	    Скаляр — как раньше. Карта — точный id модели → рантайм → ``default`` → None.
  576	    ``model`` ожидается уже канонизованным (``resolve_model``), но alias тоже переживёт.
  577	    """
  578	    if isinstance(effort, str):
  579	        return effort or None
  580	    if not isinstance(effort, dict):
  581	        return None
  582	    for key in (_canon_model(model), runtime, "default"):
  583	        val = effort.get(key)
  584	        if val:
  585	            return val
  586	    return None

  616	def build_prompt_modules(pipeline_name: str, modules: list[str] | tuple[str, ...]) -> str:
  617	    """Assemble named modules from the canonical pipeline prompt library."""
  618	    parts: list[str] = []
  619	    for module in modules:
  620	        if not _is_safe_rel(module):
  621	            raise ValueError(f"unsafe module name '{module}' (abs или '..')")
  622	        path = prompt_path(pipeline_name, f"modules/{module}.md")
  623	        if not path.is_file():
  624	            raise FileNotFoundError(
  625	                f"pipeline '{pipeline_name}' module '{module}' not found at {path}"
  626	            )
  627	        parts.append(path.read_text().strip())
  628	    return "\n\n".join(parts)

  631	def build_system_prompt(pipeline_name: str, role: str, scope: str = "") -> str:
  632	    """Собрать system_prompt из prompt_layers резолвнутой роли.
  633	
  634	    Каждый слой читается из ``.orchestra/pipelines/<name>/prompts/<layer>`` через
  635	    :func:`prompt_path` (единственный источник — app/prompts/ удалён). Отсутствующий
  636	    слой-файл пропускается. Конкатенация через ``\\n\\n``. Динамика (каталог ролей,
  637	    блоки других оркестраторов/воркеров) добавляется вызывающим в manager — здесь
  638	    только статика из файлов.
  639	
  640	    После слоёв роли инлайнятся ``modules`` — переиспользуемые блоки промпта из
  641	    ``prompts/modules/{m}.md`` (та же изоляция). Отсутствующий модуль пропускается с
  642	    warning (роль не должна падать из-за недостающего блока).
  643	
  644	    :raises FileNotFoundError: если манифест пайплайна отсутствует (на Этапе 3 manager
  645	        ловит и делегирует в legacy-путь апстрима).
  646	    """
  647	    rr = resolve_role(load_pipeline(pipeline_name), role)
  648	    parts: list[str] = []
  649	    for layer in rr.prompt_layers:
  650	        p = prompt_path(pipeline_name, layer)
  651	        if p.is_file():
  652	            parts.append(p.read_text())
  653	    module_text = build_prompt_modules(pipeline_name, rr.modules)
  654	    if module_text:
  655	        parts.append(module_text)
  656	    return "\n\n".join(parts)

  677	# ── Валидация спавна (fail-closed / fail-open) ────────────────────────────
  678	
  679	def validate_spawn(pipeline_name: str, parent_role: str | None, child_role: str) -> None:
  680	    """Проверить допустимость спавна ``child_role`` родителем ``parent_role``.
  681	
  682	    Неизвестная роль родителя или ребёнка — всегда ``ValueError``, в том числе
  683	    при ``validation: fail-open``. Иначе whitelist ``can_spawn`` действует
  684	    только на роли из манифеста, а любая чужая строка обходит его целиком (#36).
  685	
  686	    ``can_spawn: ['*']`` значит любую *известную* роль, не любую строку.
  687	    Корень (parent пуст/None) — спавн от юзера/UI, всегда разрешён. Пустой
  688	    ``child_role`` (генерик-воркер) разрешён только если ``allow_unrouted_workers``.
  689	
  690	    :raises ValueError: при запрещённом спавне.
  691	    :raises FileNotFoundError: если манифест отсутствует (manager ловит → fallback).
  692	    """
  693	    cfg = load_pipeline(pipeline_name)
  694	
  695	    if not parent_role:  # корень — спавн от юзера/UI
  696	        return
  697	    parent = cfg.roles.get(parent_role)
  698	    if parent is None:
  699	        raise ValueError(
  700	            f"unknown parent role '{parent_role}' in pipeline '{pipeline_name}'. "
  701	            f"known={known_roles(pipeline_name)}")
  702	    if not child_role:  # генерик-воркер (роль не указана)
  703	        if not parent.allow_unrouted_workers:
  704	            raise ValueError(
  705	                f"role '{parent_role}' must specify child role. "
  706	                f"allowed: {parent.can_spawn or '(none)'}")
  707	        return
  708	    if child_role not in cfg.roles:
  709	        raise ValueError(
  710	            f"unknown role '{child_role}' in pipeline '{pipeline_name}'. "
  711	            f"known={known_roles(pipeline_name)}")
  712	    if "*" in parent.can_spawn:
  713	        return
  714	    if child_role not in parent.can_spawn:
  715	        raise ValueError(
  716	            f"role '{parent_role}' cannot spawn '{child_role}'. "
  717	            f"allowed: {parent.can_spawn or '(none — terminal)'}")
