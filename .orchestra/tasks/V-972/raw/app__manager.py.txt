   93	class LockBusy(RuntimeError):
   94	    """Лок сессии занят дольше отведённого: у вызова есть кто уступить и кому повторить."""
   95	
   96	    def __init__(self, what: str, worker: str, waited: float):
   97	        self.what, self.worker, self.waited = what, worker, waited
   98	        super().__init__(
   99	            f"worker '{worker}' is busy: another operation has held the session lock for "
  100	            f"{waited:.0f}s (message delivery or merge in flight). Nothing was changed — "
  101	            f"retry {what} in ~30s."
  102	        )

  125	@asynccontextmanager
  126	async def wait_for_session_lock(lock, *, what: str, worker: str, limit: float | None = None):
  127	    """Взять лок сессии так, чтобы ожидание было ВИДНО снаружи.
  128	
  129	    Замер #27: доставка держит лок секундами (коннект CLI-бэкенда 1.6–2.0 с, а `connect()`
  130	    внутри разрешает себе до 60 с), git при этом занимает 0.01 с. Раньше ожидание не
  131	    оставляло ни строки в журнале и ни поля в ответе — снаружи оно было неотличимо от
  132	    зависшего сервера.
  133	    """
  134	    # Предел читается в момент вызова, а не как значение по умолчанию: значение по умолчанию
  135	    # вычисляется при определении функции, и подмена константы в тесте молча не сработала бы.
  136	    limit = LOCK_WAIT_LIMIT_SECONDS if limit is None else limit
  137	    started = time.monotonic()
  138	    try:
  139	        await asyncio.wait_for(lock.acquire(), timeout=limit)
  140	    except asyncio.TimeoutError:
  141	        waited = time.monotonic() - started
  142	        logger.warning("%s for %s gave up after %.1fs waiting for the session lock",
  143	                       what, worker, waited)
  144	        raise LockBusy(what, worker, waited) from None
  145	    waited = time.monotonic() - started
  146	    if waited > 0.5:
  147	        logger.warning("%s for %s waited %.1fs for the session lock", what, worker, waited)
  148	    try:
  149	        yield waited
  150	    finally:
  151	        lock.release()

  154	def next_adhoc_branch(worker_name: str) -> str:
  155	    """Имя ветки для авто-переключения перед доставкой сообщения.
  156	
  157	    Секунды эпохи БЕЗ усечения. Прежняя форма `str(int(time.time()))[-6:]` замыкала
  158	    пространство имён каждые 10**6 с = 11.57 суток, а ветки `adhoc-*` не удаляются —
  159	    столкновение было гарантировано временем, а не вероятностью, и приводило к тихому
  160	    переселению воркера на чужую ветку 11-суточной давности (#27, E2).
  161	
  162	    ВНИМАНИЕ, зависимость между слоями: порядковый номер живёт в ПРОЦЕССЕ и обнуляется
  163	    рестартом, поэтому теоретически возможен повтор имени при рестарте внутри одной
  164	    секунды. Это допустимо ТОЛЬКО потому, что второй слой — `expect_absent=True` в
  165	    `switch_worktree_branch` — делает такой повтор громким отказом. Снимете второй слой
  166	    как «лишнюю паранойю» — вернёте тихую порчу данных, а не просто упростите код.
  167	    """
  168	    return f"adhoc-{int(time.time())}-{next(_adhoc_serial)}/{worker_name}"

  181	async def _wait_owned_task(task: asyncio.Task) -> asyncio.CancelledError | None:
  182	    """Wait without letting caller cancellation penetrate an owned operation."""
  183	    cancellation = None
  184	    current = asyncio.current_task()
  185	    while not task.done():
  186	        try:
  187	            await asyncio.shield(task)
  188	        except asyncio.CancelledError as error:
  189	            if current is not None and current.cancelling():
  190	                cancellation = cancellation or error
  191	                while current.cancelling():
  192	                    current.uncancel()
  193	            elif task.done():
  194	                break
  195	            else:
  196	                raise
  197	        except BaseException:
  198	            break
  199	    return cancellation

  206	def _fmt_role_catalog_entry(rr) -> str:
  207	    """Форматировать одну запись каталога ролей из :class:`ResolvedRole`.
  208	
  209	    Совпадает по форме с ``_roles_catalog`` (заголовок ### `name` (label) — model,
  210	    описание, when/not_for). Источник полей — манифест (ResolvedRole), не frontmatter.
  211	    """
  212	    desc = (rr.description or "").strip().replace("\n", " ")
  213	    entry = f"### `{rr.name}` ({rr.label}) — model: {rr.model}"
  214	    if desc:
  215	        entry += f"\n{desc}"
  216	    if rr.when:
  217	        entry += f"\n- ✅ **When**: {rr.when.strip()}"
  218	    if rr.not_for:
  219	        entry += f"\n- ❌ **Not for**: {rr.not_for.strip()}"
  220	    if "*" in rr.can_spawn:
  221	        spawn = "any role"
  222	    elif rr.can_spawn:
  223	        spawn = ", ".join(f"`{r}`" for r in rr.can_spawn)
  224	    else:
  225	        spawn = "none (cannot spawn)"
  226	    entry += f"\n- 🌱 **Can spawn**: {spawn}"
  227	    skills = rr.skills
  228	    if isinstance(skills, list) and skills:
  229	        entry += f"\n- 🔧 **Skills**: {', '.join(skills)}"
  230	    return entry

  233	def _roles_catalog_from_manifest(pipeline: str, parent_role: str) -> str:
  234	    """Каталог ролей оркестратору из манифеста, отфильтрованный по ``can_spawn``.
  235	
  236	    B2: показываем ВСЕ роли из ``can_spawn`` родителя (включая под-оркестраторов).
  237	    ``can_spawn=['*']`` → все роли пайплайна. Сортировка по ``order``. Закрывает
  238	    дефект плоского ``_roles_catalog`` (показывал бы запретные роли).
  239	    """
  240	    cfg = load_pipeline(pipeline)
  241	    parent = cfg.roles.get(parent_role)
  242	    if parent is None:
  243	        return ""
  244	    if "*" in parent.can_spawn:
  245	        # S1: wildcard НЕ включает саму роль-родителя (upstream _roles_catalog
  246	        # пропускал orchestrator из своего же каталога воркеров).
  247	        visible = [r for r in cfg.roles if r != parent_role]
  248	    else:
  249	        visible = list(parent.can_spawn)
  250	    visible = [r for r in visible if r in cfg.roles]
  251	    if not visible:
  252	        return ""
  253	    entries = [
  254	        _fmt_role_catalog_entry(resolve_role(cfg, r))
  255	        for r in sorted(visible, key=lambda r: cfg.roles[r].order)
  256	    ]
  257	    return ('## Available worker roles\nSpawn with `role="<name>"`. '
  258	            'If no role specified, defaults to `worker`.\n\n' + "\n\n".join(entries))

  261	def ROLE_SYSTEM_PROMPT(pipeline: str, role: str, scope: str = "") -> str:
  262	    """Системный промпт роли из файлов: слои, индекс KB и статический каталог ролей.
  263	
  264	    Единственный источник — ``.orchestra/pipelines/<pipeline>/prompts/`` через
  265	    :func:`build_system_prompt`. Для оркестратора добавляется каталог ролей
  266	    (фильтр ``can_spawn``). Живое состояние доступно через инструменты и API,
  267	    поэтому изменения БД не переписывают кешируемый префикс.
  268	
  269	    Оглавление базы знаний ``scope`` идёт ЗДЕСЬ, а не в файлах ролей: список тем
  270	    принадлежит проекту агента, а не пайплайну, и обязан обновляться без рестарта —
  271	    эта функция зовётся и при спавне, и при переинжекте (:meth:`assemble_prompt`).
  272	    Блока нет ни в одном ``prompts/modules/*.md``, поэтому роль с коротким списком
  273	    ``modules`` (reducer) не может его потерять — на этом #490 терял вынесенные блоки.
  274	
  275	    Fail loud: нет манифеста или роли нет в манифесте → :class:`ValueError`.
  276	    Legacy-fallback на ``app/prompts/`` удалён (единый источник = pipelines).
  277	    """
  278	    try:
  279	        base = build_system_prompt(pipeline, role, scope)
  280	    except (FileNotFoundError, KeyError) as e:
  281	        raise ValueError(
  282	            f"role '{role}' not resolvable in pipeline '{pipeline}': {e!r}. "
  283	            f"Define it in .orchestra/pipelines/{pipeline}/pipeline.yaml + prompts/roles/{role}.md"
  284	        ) from e
  285	    kb = kb_index_block(scope)
  286	    if kb:
  287	        base += f"\n\n{kb}"
  288	    rr = get_role(pipeline, role)
  289	    is_orch = rr.is_orchestrator if rr is not None else is_orchestrator_role(role)
  290	    if is_orch:
  291	        catalog = _roles_catalog_from_manifest(pipeline, role)
  292	        if catalog:
  293	            base += f"\n\n{catalog}"
  294	    return base

  337	def _parse_custom_mcp(raw) -> dict:
  338	    """Sanitize custom MCP servers (from DB JSON string or a dict).
  339	    Returns a dict with the `orchestra` key stripped. Non-dict input -> {}."""
  340	    if not raw:
  341	        return {}
  342	    if isinstance(raw, str):
  343	        import json
  344	        try:
  345	            raw = json.loads(raw)
  346	        except (json.JSONDecodeError, TypeError):
  347	            logger.warning("invalid mcp_servers_custom JSON; ignoring")
  348	            return {}
  349	    if not isinstance(raw, dict):
  350	        logger.warning(f"mcp_servers_custom is not an object ({type(raw).__name__}); ignoring")
  351	        return {}
  352	    return {k: v for k, v in raw.items() if k != "orchestra"}

  355	def _make_mcp_config(name: str, scope: str, role: str = "worker",
  356	                     parent_name: str = "", extra: dict | None = None,
  357	                     session_id: str = "", pipeline: str = "",
  358	                     disabled_tools: list[str] | None = None) -> dict:
  359	    role_spec = get_role(pipeline or get_active_pipeline(scope), role)
  360	    denied = parse_disabled_tools(disabled_tools)
  361	    denied = sorted(set(denied) | set(role_spec.disabled_tools if role_spec else []))
  362	    env = {
  363	        "ORCHESTRA_DISABLED_TOOLS": json.dumps(denied),
  364	        **MCP_BASE_ENV,
  365	        "ORCHESTRA_URL": "http://127.0.0.1:8888",
  366	        "ORCHESTRA_SCOPE": scope,
  367	        "ORCHESTRA_ROLE": role,
  368	        "ORCHESTRA_ACCESS_MODE": "reducer" if role == "reducer" else "full",
  369	        "WORKER_NAME": name,
  370	        "PARENT_NAME": parent_name,
  371	        # Имя агент может сменить, id — нет. Всё, что СОХРАНЯЕТСЯ и сравнивается позже
  372	        # (держатель тест-лока), обязано опираться на него, а не на имя (#82).
  373	        "ORCHESTRA_SESSION_ID": session_id,
  374	    }
  375	    from app.mcp_proof import PROOF_ENV, issue_mcp_proof
  376	
  377	    proof = issue_mcp_proof(session_id)
  378	    if proof:
  379	        env[PROOF_ENV] = proof
  380	    cfg = {"orchestra": {"command": MCP_STDIO_CMD[0], "args": MCP_STDIO_CMD[1:], "env": env, "alwaysLoad": True}}
  381	    if extra:

  383	            if k == "orchestra":
  384	                logger.warning("custom MCP server 'orchestra' would override Orchestra MCP — ignored")
  385	                continue
  386	            cfg[k] = v
  387	    return cfg

  405	OWNERSHIP_MARKER = "\n\n## Directory ownership\n"
  406	# Opening tag only: the number of newlines before it varies between the spawn path and
  407	# refresh_worker_memory (app/prompting.py:27 matches it as `\n*<worker-memory>`).
  408	WORKER_MEMORY_MARKER = "<worker-memory>"
  409	
  410	
  411	def ownership_block(owned_dirs: list[str]) -> str:
  412	    """The generated directory-ownership suffix. Single owner of its wording."""
  413	    if not owned_dirs:
  414	        return ""
  415	    lines = "\n".join(f"- {d}/" for d in owned_dirs)
  416	    return (OWNERSHIP_MARKER +
  417	            "Expected work areas (advisory, not an edit allowlist):\n"
  418	            f"{lines}\n"
  419	            "Edit other files needed for the approved task within your worktree. "
  420	            "Respect explicit task exclusions and coordinate overlapping changes.")
  421	
  422	
  423	def replace_ownership_block(
  424	    prompt: str, before: list[str], after: list[str],
  425	) -> tuple[str, bool]:
  426	    """Swap the generated ownership block for the one ``after`` produces.
  427	
  428	    Ownership lives in two places — the ``sessions.owned_dirs`` column and this block
  429	    inside the stored prompt — so a migration that fixes only the column leaves the
  430	    worker reading the old paths.
  431	
  432	    The block is matched WHOLE, reconstructed from ``before``, not located by its
  433	    heading: operator task text may quote "## Directory ownership" followed by a
  434	    bullet, and a heading search would rewrite that quotation while leaving the real
  435	    generated suffix untouched. Returns ``(prompt, False)`` when the exact block is
  436	    absent, so the caller can report a drifted prompt instead of guessing at it.
  437	    """
  438	    old_block = ownership_block(before)
  439	    if not old_block:
  440	        return prompt, False
  441	    # Worker memory is appended AFTER the ownership suffix and is written by the worker
  442	    # itself, so it can quote a whole ownership block. Searching the full prompt would let
  443	    # `rpartition` rewrite that quotation and leave the real block — the one assemble_prompt
  444	    # keeps — untouched. Only the region before the memory can hold the generated block.
  445	    body, memory_separator, memory = prompt.partition(WORKER_MEMORY_MARKER)
  446	    if old_block not in body:
  447	        # Upgrade only the exact historical generated suffix, never free-form exclusions.
  448	        lines = "\n".join(f"- {d}/" for d in before)
  449	        old_block = (OWNERSHIP_MARKER +
  450	                     "You OWN these directories — edit ONLY files under them:\n"
  451	                     f"{lines}\n"
  452	                     "Do NOT touch files outside your owned directories. "
  453	                     "If the task requires it — STOP and ask the orchestrator.")
  454	        if old_block not in body:
  455	            return prompt, False
  456	    head, _, tail = body.rpartition(old_block)
  457	    return head + ownership_block(after) + tail + memory_separator + memory, True

  460	_RESTART_REPLAY_LIMIT = 20_000
  461	
  462	
  463	def restart_notice_text(row: dict) -> str:
  464	    """Wake-up text for an agent whose turn a restart cut off.
  465	
  466	    Without a native session id the runtime starts an empty conversation, so the messages of
  467	    the cut turn are replayed: otherwise the agent wakes up without its task (V-919).
  468	    """
  469	    text = (
  470	        "[system] Orchestra server restarted while your turn was running. "
  471	        "Commands running inside the turn were stopped; background jobs created with "
  472	        "bg_create keep running and report as usual. Continue your task from where you "
  473	        "stopped: check files and external action outcomes first and do not blindly repeat "
  474	        "the last operation."
  475	    )
  476	    if row.get("session_id"):
  477	        return text + " The saved conversation was restored."
  478	    from app.db import _conn
  479	    with _conn() as c:
  480	        last_end = c.execute(
  481	            "SELECT COALESCE(MAX(id), 0) FROM logs WHERE session_id=? AND type='status' "
  482	            "AND content LIKE 'turn ended%'", (row["id"],),
  483	        ).fetchone()[0]
  484	        messages = [r[0] for r in c.execute(
  485	            "SELECT content FROM logs WHERE session_id=? AND type='user_message' AND id>? "
  486	            "ORDER BY id", (row["id"], last_end),
  487	        ).fetchall()]
  488	    if not messages:
  489	        return text
  490	    replay = "\n\n---\n\n".join(messages)
  491	    if len(replay) > _RESTART_REPLAY_LIMIT:
  492	        replay = "…" + replay[-_RESTART_REPLAY_LIMIT:]
  493	    return (
  494	        text + " The conversation could not be restored because the turn was cut before "
  495	        "it was saved. Messages of that turn, oldest first:\n\n" + replay
  496	    )

 1296	    async def change_orchestrator_scope(self, name: str, old_scope: str,
 1297	                                         new_scope: str, new_cwd: str) -> dict:
 1298	        """Move an idle, worker-free orchestrator to a new scope/cwd.
 1299	
 1300	        Updates DB (db.change_scope), then rebuilds the runtime: kills the old MCP
 1301	        subprocess via _disconnect_backend(), swaps scope/cwd/mcp_servers, and lets
 1302	        the backend lazily reconnect on the next send() with the new ORCHESTRA_SCOPE.
 1303	        session_id is preserved → context survives.
 1304	        """
 1305	        old_scope = old_scope.rstrip("/")
 1306	        new_scope = new_scope.rstrip("/")
 1307	        new_cwd = new_cwd.rstrip("/")
 1308	        session = self.get_by_name(name, old_scope)
 1309	        # live-only: detached hydrate has no backend/MCP to rebuild
 1310	        if not session or not session.loaded:
 1311	            return {"error": f"orchestrator '{name}' not loaded in scope '{old_scope}'"}
 1312	        if not session.is_orchestrator:
 1313	            return {"error": f"'{name}' is not an orchestrator — scope change is orchestrator-only"}
 1314	        if not Path(new_cwd).is_dir():
 1315	            return {"error": f"new_cwd does not exist: {new_cwd}"}
 1316	
 1317	        live_workers = self._live_workers_in_scope(old_scope)
 1318	        if live_workers:
 1319	            return {"error": f"cannot change scope: live workers in '{old_scope}' — "
 1320	                             f"merge+kill first: {', '.join(live_workers)}"}
 1321	
 1322	        # Hold the session lifecycle lock so a concurrent send() cannot flip
 1323	        # the session IDLE→RUNNING between the idle check and the disconnect.
 1324	        # (send() only starts a fresh turn inside this same lock.)
 1325	        async with session._lifecycle_lock:
 1326	            if session.status.value == "running":
 1327	                return {"error": "cannot change scope while running — wait for idle"}
 1328	
 1329	            # Re-check under the lock right before the DB write to shrink the
 1330	            # worker-spawn TOCTOU window (a spawn could have landed since the
 1331	            # pre-lock scan). Full closure needs a scope-level spawn lock.
 1332	            live_workers = self._live_workers_in_scope(old_scope)
 1333	            if live_workers:
 1334	                return {"error": f"cannot change scope: live workers appeared in '{old_scope}' — "
 1335	                                 f"merge+kill first: {', '.join(live_workers)}"}
 1336	
 1337	            # Stop the backend (no new persists from this session) and drain any
 1338	            # in-flight _persist() BEFORE the transaction, so change_scope()'s
 1339	            # synchronous scope+cwd write is the last writer. Otherwise a stale
 1340	            # queued persist (snapshot cwd=/old) could land after the transaction
 1341	            # and clobber cwd, leaving scope=/new + cwd=/old on disk.
 1342	            await session._disconnect_backend()
 1343	            await session._drain_persist()
 1344	
 1345	            from app.db import change_scope
 1346	            result = change_scope(session.id, old_scope, new_scope, new_cwd)
 1347	            if not result.get("ok"):
 1348	                return result
 1349	
 1350	            session.scope = new_scope
 1351	            session.cwd = new_cwd
 1352	            # Migrate CLI session file so resume works in new cwd
 1353	            if session.session_id:
 1354	                self._migrate_cli_session(session.session_id, old_scope, new_scope)
 1355	            session.mcp_servers = _make_mcp_config(name, new_scope, session.role,
 1356	                                                   parent_name=session.parent_name,
 1357	                                                   extra=session.mcp_servers_custom,
 1358	                                                   session_id=session.id, pipeline=session.pipeline,
 1359	                                                   disabled_tools=session.disabled_tools)
 1360	            session._persist()
 1361	            if session._persist_task:
 1362	                await asyncio.gather(session._persist_task, return_exceptions=True)
 1363	        logger.info(f"Orchestrator '{name}' scope changed: {old_scope} → {new_scope}")
 1364	        return result

 1515	    def update_session_fields(self, name: str, scope: str, **fields) -> AgentSession | None:
 1516	        """Update simple session fields atomically for the caller.
 1517	        Live session → setattr + _persist(); detached → direct DB UPDATE."""
 1518	        unknown = set(fields) - self._UPDATABLE_FIELDS
 1519	        if unknown:
 1520	            raise ValueError(f"non-updatable fields: {sorted(unknown)}")
 1521	        found = self.get_by_name(name, scope)
 1522	        if not found:
 1523	            return None
 1524	        if found.loaded:
 1525	            for k, v in fields.items():
 1526	                setattr(found, k, v)
 1527	            if "system_prompt" in fields:
 1528	                # The endpoint replaces the complete assembled prompt, so no older
 1529	                # separated overlay may override it on the next reload.
 1530	                found.prompt_overlay = None
 1531	                found._current_prompt = fields["system_prompt"]
 1532	            found._persist()
 1533	        else:
 1534	            from app.db import _conn
 1535	            db_fields = dict(fields)
 1536	            if "system_prompt" in fields:
 1537	                db_fields["prompt_overlay"] = None
 1538	            sets = ", ".join(f"{k}=?" for k in db_fields)
 1539	            vals = [int(v) if isinstance(v, bool) else v for v in db_fields.values()]
 1540	            with _conn() as c:
 1541	                cur = c.execute(
 1542	                    f"UPDATE sessions SET {sets} WHERE id=?", (*vals, found.id),
 1543	                )
 1544	                if cur.rowcount == 0:
 1545	                    # Молчаливый ноль здесь означал бы, что поля «сохранены» только
 1546	                    # в памяти вызывающего, а в БД их нет (#54).
 1547	                    logger.warning(
 1548	                        "UPDATE sessions changed 0 rows: fields %s for id=%r not persisted",
 1549	                        list(fields), found.id,
 1550	                    )
 1551	        return found

 1587	    async def transition_lifecycle(
 1588	        self,
 1589	        session: AgentSession,
 1590	        *,
 1591	        branch: str,
 1592	        base_branch: str,
 1593	        task_id: str,
 1594	        needs_switch: bool,
 1595	        owned_dirs: list[str] | None = None,
 1596	    ) -> None:
 1597	        """Atomically publish a successful branch/task transition.
 1598	
 1599	        Git is deliberately completed by the caller first. Until this DB transaction
 1600	        succeeds, the in-memory and durable lifecycle remain untouched. ``owned_dirs``
 1601	        is ``None`` for same-task continuation and a list (including ``[]``) for a new
 1602	        task, which also replaces the ownership prompt.
 1603	        """
 1604	        # A few lifecycle callers use lightweight detached doubles in tests and
 1605	        # integrations. They have no full session snapshot; retain the old lifecycle
 1606	        # seam for those data-only objects while real AgentSession transitions use one
 1607	        # complete save_session transaction below.
 1608	        if not hasattr(session, "_to_db_dict"):
 1609	            await self.persist_lifecycle(
 1610	                session,
 1611	                branch=branch,
 1612	                base_branch=base_branch,
 1613	                task_id=task_id,
 1614	                needs_switch=needs_switch,
 1615	            )
 1616	            return
 1617	        if session.loaded:
 1618	            await session._drain_persist()
 1619	        snapshot = session._to_db_dict()
 1620	        snapshot.update(
 1621	            branch=branch,
 1622	            base_branch=base_branch,
 1623	            task_id=task_id,
 1624	            needs_switch=int(needs_switch),
 1625	        )
 1626	        new_prompt = None
 1627	        new_overlay = session.prompt_overlay
 1628	        normalized = None
 1629	        if owned_dirs is not None:
 1630	            normalized = parse_owned_dirs(owned_dirs)
 1631	            new_prompt, new_overlay = self._transition_prompt(session, normalized)
 1632	            snapshot.update(
 1633	                owned_dirs=json.dumps(normalized) if normalized else "",
 1634	                system_prompt=new_prompt,
 1635	                prompt_overlay=new_overlay,
 1636	            )
 1637	        await asyncio.to_thread(save_session, snapshot)
 1638	
 1639	        session.branch = branch
 1640	        session.base_branch = base_branch
 1641	        session.task_id = task_id
 1642	        session.needs_switch = needs_switch
 1643	        if normalized is not None:
 1644	            session.owned_dirs = normalized
 1645	            session.system_prompt = new_prompt or ""
 1646	            session.prompt_overlay = new_overlay
 1647	            session._current_prompt = new_prompt or ""
 1648	            # The next turn must deliver the new ownership prompt at the idle boundary.
 1649	            session._prompt_injected = False
 1650	        if session.db_row is not None:
 1651	            session.db_row.update(
 1652	                branch=branch,
 1653	                base_branch=base_branch,
 1654	                task_id=task_id,
 1655	                needs_switch=int(needs_switch),
 1656	            )
 1657	            if normalized is not None:
 1658	                session.db_row.update(
 1659	                    owned_dirs=json.dumps(normalized) if normalized else "",
 1660	                    system_prompt=new_prompt or "",
 1661	                    prompt_overlay=new_overlay,
 1662	                )

 1690	    async def apply_owned_dirs(
 1691	        self, session: AgentSession, owned_dirs: list[str],
 1692	    ) -> list[str]:
 1693	        """Replace ownership without touching the branch or the task.
 1694	
 1695	        Both owners move together — the ``owned_dirs`` column and the ownership
 1696	        block inside the stored prompt — because a column-only write leaves the
 1697	        worker reading its previous directories. An empty list clears ownership.
 1698	        The caller checks the status: rewriting the prompt of a running turn would
 1699	        change the contract under a worker that is already editing files.
 1700	        """
 1701	        # Validation reads every live worker of this scope, so validate+write must be one
 1702	        # critical section: two concurrent calls for DIFFERENT workers each lock only their
 1703	        # own session, both pass against the pre-write state, and both commit the same
 1704	        # directory. (The same race still exists on the older spawn/switch paths — those
 1705	        # take only a per-name lock — and closing it there is a separate change.)
 1706	        async with self._scope_ownership_lock(session.scope):
 1707	            normalized = self.validate_owned_dirs_transition(session, owned_dirs)
 1708	            if session.loaded:
 1709	                await session._drain_persist()
 1710	            new_prompt, new_overlay = self._transition_prompt(session, normalized)
 1711	            stored_dirs = json.dumps(normalized) if normalized else ""
 1712	
 1713	            async def _commit() -> None:
 1714	                """Persist and publish as one step that a cancelled request cannot split.
 1715	
 1716	                A client disconnect between the DB write and the in-memory publication
 1717	                would leave SQLite holding the new boundary while the live worker keeps
 1718	                the old one — and release the locks that were supposed to protect exactly
 1719	                that gap.
 1720	                """
 1721	                if session.loaded:
 1722	                    snapshot = session._to_db_dict()
 1723	                    snapshot.update(
 1724	                        owned_dirs=stored_dirs,
 1725	                        system_prompt=new_prompt,
 1726	                        prompt_overlay=new_overlay,
 1727	                    )
 1728	                    await asyncio.to_thread(save_session, snapshot)
 1729	                else:
 1730	                    # A detached session is hydrated from a partial row: `_hydrate_row`
 1731	                    # never fills `color` or `template_hash`, so persisting its full
 1732	                    # `_to_db_dict()` snapshot blanks those columns for a worker nobody
 1733	                    # asked to change.
 1734	                    await asyncio.to_thread(
 1735	                        self._persist_owned_dirs_only,
 1736	                        session.id, stored_dirs, new_prompt, new_overlay,
 1737	                    )
 1738	                session.owned_dirs = normalized
 1739	                session.system_prompt = new_prompt
 1740	                session.prompt_overlay = new_overlay
 1741	                session._current_prompt = new_prompt
 1742	                # The next turn must deliver the new ownership prompt at the idle boundary.
 1743	                session._prompt_injected = False
 1744	                if session.db_row is not None:
 1745	                    session.db_row.update(
 1746	                        owned_dirs=stored_dirs,
 1747	                        system_prompt=new_prompt,
 1748	                        prompt_overlay=new_overlay,
 1749	                    )
 1750	
 1751	            # A single `shield` is not enough on two counts: the caller would leave this
 1752	            # block and release the scope, lifecycle and session locks while the write is
 1753	            # still in flight, and a SECOND cancellation penetrates a plain `await commit`
 1754	            # — cancelling the `to_thread` future while the SQLite thread keeps going and
 1755	            # commits, leaving the DB ahead of the live session. `_wait_owned_task`
 1756	            # absorbs repeated cancellation and returns it to re-raise afterwards.
 1757	            commit = asyncio.create_task(_commit())
 1758	            cancellation = await _wait_owned_task(commit)
 1759	            commit.result()
 1760	            if cancellation is not None:
 1761	                raise cancellation
 1762	        return normalized

 1787	    def _resolve_base_branch(self, base_branch: str, pipeline: str, role: str,
 1788	                             parent_name: str, scope: str, repo_path: str) -> str:
 1789	        """Резолв базовой ветки worktree по стратегии манифеста (DESIGN §10, B3).
 1790	
 1791	        Explicit base wins. ``strategy=parent`` uses the parent's actual branch;
 1792	        ``strategy=main`` and missing parents use the repository's verifiable mainline.
 1793	        """
 1794	        if base_branch:
 1795	            return resolve_git_base_branch(repo_path, base_branch)
 1796	        try:
 1797	            rr = get_role(pipeline, role)
 1798	        except FileNotFoundError:
 1799	            rr = None
 1800	        if rr is None or rr.base_branch_strategy == "main":
 1801	            return resolve_git_base_branch(repo_path)
 1802	        parent_branch = ""
 1803	        parent_dir = ""
 1804	        if parent_name:
 1805	            ps = self.get_by_name(parent_name, scope)
 1806	            if ps is not None:
 1807	                parent_branch = ps.branch or ""
 1808	                parent_dir = ps.worktree_path or ps.cwd or ""
 1809	        if not parent_branch:
 1810	            logger.warning(
 1811	                "base_branch_strategy=parent, но у родителя '%s' нет ветки — "
 1812	                "resolving repository mainline",
 1813	                parent_name)
 1814	            return resolve_git_base_branch(repo_path)
 1815	        try:
 1816	            return resolve_git_base_branch(repo_path, parent_branch)
 1817	        except ValueError:
 1818	            # Ветка родителя может жить в ДРУГОМ репозитории: в одном scope их бывает
 1819	            # несколько (у seedon `site/` и `infra/` — свои git root'ы со своими origin).
 1820	            # «Ответвись от родителя» через границу репозитория не значит ничего — там
 1821	            # другая история. Берём mainline репозитория РЕБЁНКА, но громко: молчаливая
 1822	            # подмена базы хуже отказа. Если репозиторий ТОТ ЖЕ, отсутствие ветки —
 1823	            # настоящая ошибка, и она летит дальше (#69).
 1824	            if parent_dir and _crosses_repo_boundary(parent_dir, repo_path):
 1825	                logger.warning(
 1826	                    "base_branch_strategy=parent: ветка '%s' родителя '%s' живёт в другом "
 1827	                    "репозитории (%s), воркер спавнится в %s — база взята из mainline его "
 1828	                    "репозитория",
 1829	                    parent_branch, parent_name, parent_dir, repo_path)
 1830	                return resolve_git_base_branch(repo_path)
 1831	            raise

 1880	    async def _ensure_loaded_row(self, db_row: dict) -> AgentSession:
 1881	        session_id = db_row["id"]
 1882	        session = self.sessions.get(session_id)
 1883	        if session is not None:
 1884	            return session
 1885	        async with self.get_session_lock(session_id):
 1886	            session = self.sessions.get(session_id)
 1887	            if session is not None:
 1888	                return session
 1889	            # Re-read under the lock: `db_row` was fetched by the caller BEFORE the lock,
 1890	            # and a writer holding it meanwhile (ownership, prompt, model) has already
 1891	            # committed. Hydrating the captured row would publish a live session that
 1892	            # silently disagrees with SQLite — for ownership that means a worker running
 1893	            # under a boundary the database says it no longer has.
 1894	            return await self._load_from_db(get_session(session_id) or db_row)

 1896	    def assemble_prompt(
 1897	        self, *, pipeline: str, role: str, scope: str, is_orch: bool, name: str,
 1898	        owned_dirs, branch: str, stored_overlay: str | None, old_prompt: str,
 1899	        repository_path: str = "", parent_name: str | None = None,
 1900	        allow_missing_layout: bool = False,
 1901	    ) -> tuple[str, str | None]:
 1902	        """Собрать системный промпт из файлов ролей — один владелец на двух вызывающих.
 1903	
 1904	        Зовётся из `_load_from_db` (восстановление сессии при старте) и из переинжекта
 1905	        в `session.py`: без второго вызывающего правка `.orchestra/pipelines/**` доезжала до живого
 1906	        агента только рестартом (#220 T1, медиана задержки выката 3.3 ч).
 1907	
 1908	        Возвращает (собранный промпт, восстановленный overlay). `overlay is None` на
 1909	        выходе означает полную замену промпта оператором — у неё нет границы
 1910	        компонентов, и пересобирать её нельзя.
 1911	        """
 1912	        current_base = ROLE_SYSTEM_PROMPT(pipeline, role, scope)
 1913	        if not is_orch:
 1914	            current_base = safe_format_prompt(
 1915	                current_base,
 1916	                worker_name=name, orchestrator_name=parent_name or "orchestrator",
 1917	                scope=scope, branch=branch,
 1918	            )
 1919	        if stored_overlay is None:
 1920	            old_without_memory = temp_rules.strip_block(strip_worker_memory(old_prompt))
 1921	            if not old_without_memory:
 1922	                prompt_overlay = self._ownership_prompt(parse_owned_dirs(owned_dirs))
 1923	                prompt_without_memory = current_base + prompt_overlay
 1924	            elif old_without_memory.startswith(current_base):
 1925	                prompt_overlay = old_without_memory[len(current_base):]
 1926	                prompt_without_memory = current_base + prompt_overlay
 1927	            elif (
 1928	                "<role>" in old_without_memory
 1929	                and "</role>" in old_without_memory
 1930	                and (
 1931	                    ("<knowledge>" in old_without_memory and "</knowledge>" in old_without_memory)
 1932	                )
 1933	            ):
 1934	                # Pre-overlay pipeline prompts are identifiable by their complete platform
 1935	                # envelopes. They are not operator overrides: rebuild the current base while
 1936	                # retaining the separately stored directory-ownership contract. A free-form full
 1937	                # prompt has no such boundary and remains byte-preserved in the branch below.
 1938	                prompt_overlay = self._ownership_prompt(parse_owned_dirs(owned_dirs))
 1939	                prompt_without_memory = current_base + prompt_overlay
 1940	            else:
 1941	                # A legacy/full-prompt override has no component boundary to recover.
 1942	                # Preserve it byte-for-byte instead of guessing and dropping authority.
 1943	                prompt_overlay = None
 1944	                prompt_without_memory = old_without_memory
 1945	        else:
 1946	            prompt_overlay = safe_format_prompt(
 1947	                strip_worker_memory(stored_overlay), worker_name=name,
 1948	                orchestrator_name=parent_name or "orchestrator", scope=scope, branch=branch,
 1949	            )
 1950	            prompt_without_memory = current_base + prompt_overlay
 1951	        if prompt_overlay is not None:
 1952	            areas = parse_owned_dirs(owned_dirs)
 1953	            prompt_overlay, _ = replace_ownership_block(prompt_overlay, areas, areas)
 1954	            prompt_without_memory = current_base + prompt_overlay
 1955	        refreshed = refresh_worker_memory(
 1956	            prompt_without_memory, name, role, scope, repository_path,
 1957	            allow_absent_project=True,
 1958	            allow_missing_layout=allow_missing_layout,
 1959	        )
 1960	        # Last on purpose: the volatile block must not invalidate the cached stable prefix.
 1961	        return temp_rules.with_block(
 1962	            refreshed, role=role, is_orch=is_orch, scope=scope, name=name), prompt_overlay

 2359	    async def auto_resume_all(self) -> None:
 2360	        self.reconcile_live_colors()
 2361	        from app.db import _conn
 2362	        with _conn() as c:
 2363	            # 'interrupted' — graceful shutdown успел пометить оборванный ход;
 2364	            # 'running' — не успел (SIGKILL/OOM). Оба означают одно: ход прерван
 2365	            # рестартом. Читать только 'running' значило будить лишь тех, кого
 2366	            # застал аварийный путь (#160).
 2367	            was_running_rows = [dict(r) for r in c.execute(
 2368	                "SELECT * FROM sessions WHERE status IN ('running', 'interrupted')"
 2369	            ).fetchall()]
 2370	            was_running = {r["id"] for r in was_running_rows}
 2371	            was_waiting = {r["id"] for r in c.execute(
 2372	                "SELECT id FROM sessions WHERE status = 'waiting'"
 2373	            ).fetchall()}
 2374	            # A turn cut before the runtime reported its native id (first turn, V-919) has no
 2375	            # session_id, yet it is exactly the agent that must continue.
 2376	            resumable = [dict(r) for r in c.execute(
 2377	                "SELECT * FROM sessions WHERE (session_id IS NOT NULL "
 2378	                "AND status IN ('running', 'interrupted', 'idle', 'waiting')) "
 2379	                "OR status IN ('running', 'interrupted')"
 2380	            ).fetchall()]
 2381	            c.execute("UPDATE sessions SET status='idle' "
 2382	                      "WHERE status IN ('running', 'interrupted', 'waiting')")
 2383	
 2384	        # R1: load orchestrators first — workers need their parent_name resolved,
 2385	        # and the orchestrator's on_idle callback registered before workers resume
 2386	        orchs = [r for r in resumable if bool(r.get("is_orchestrator")) or is_orchestrator_role(r.get("role", "worker"))]
 2387	        workers = [r for r in resumable if not (bool(r.get("is_orchestrator")) or is_orchestrator_role(r.get("role", "worker")))]
 2388	        pending_handoffs = {
 2389	            row["session_id"]: row for row in list_latest_runtime_handoffs()
 2390	        }
 2391	
 2392	        not_continued = []
 2393	        for kind, rows in (("orchestrator", orchs), ("worker", workers)):
 2394	            for row in rows:
 2395	                if row["id"] in self.sessions:
 2396	                    continue
 2397	                if not Path(row.get("cwd") or row["scope"]).is_dir():
 2398	                    if row["id"] in was_running:
 2399	                        not_continued.append(row)
 2400	                    continue
 2401	                try:
 2402	                    session = await self._load_from_db(
 2403	                        row, recovery_handoff=pending_handoffs.get(row["id"]),
 2404	                    )
 2405	                    logger.info("Resumed %s: %s", kind, row["name"])
 2406	                    if row["id"] in was_waiting:
 2407	                        from app.bg_jobs import bg_manager
 2408	                        if bg_manager and bg_manager.has_active_jobs(row["id"]):
 2409	                            session.status = AgentStatus.WAITING
 2410	                            session._persist()
 2411	                    elif row["id"] in was_running:
 2412	                        spawn_supervised(
 2413	                            self._inject_restart_notice(session, row),
 2414	                            f"уведомление о рестарте для {session.name}",
 2415	                        )
 2416	                except Exception as e:
 2417	                    logger.error("Failed to resume %s %s: %s", kind, row["name"], e)
 2418	                    if row["id"] in was_running:

 2421	        # made orchestrators intervene while the worker was already continuing (V-919).
 2422	        # The continued turn reports its own end; only a turn that cannot continue is
 2423	        # signalled here.
