    65	def parse_outbox(data: bytes) -> tuple[list[dict], list[str]]:
    66	    """(valid entries, reasons for everything dropped). Never raises on hostile content."""
    67	    dropped: list[str] = []
    68	    if len(data) > MAX_OUTBOX_BYTES:
    69	        dropped.append(f"outbox larger than {MAX_OUTBOX_BYTES // 1024} KiB; the tail was ignored")
    70	        data = data[:MAX_OUTBOX_BYTES]
    71	    entries: list[dict] = []
    72	    for number, raw in enumerate(data.decode("utf-8", "replace").splitlines(), 1):
    73	        if not raw.strip():
    74	            continue
    75	        if len(entries) >= MAX_ENTRIES:
    76	            dropped.append(f"line {number}: more than {MAX_ENTRIES} entries in one turn")
    77	            break
    78	        try:
    79	            entry = json.loads(raw)
    80	        except ValueError:
    81	            dropped.append(f"line {number}: not JSON")
    82	            continue
    83	        if not isinstance(entry, dict) or not isinstance(entry.get("tool"), str):
    84	            dropped.append(f"line {number}: no tool name")
    85	            continue
    86	        args = entry.get("args", {})
    87	        if not isinstance(args, dict):
    88	            dropped.append(f"line {number}: args must be an object")
    89	            continue
    90	        if entry["tool"] not in ALLOWED_TOOLS:
    91	            dropped.append(f"line {number}: tool '{entry['tool'][:40]}' is not available to hub workers")
    92	            continue
    93	        entries.append({"tool": entry["tool"], "args": args})
    94	    return entries, dropped

   126	async def execute(entries: list[dict], who: Identity, call: ApiCall) -> OutboxReport:
   127	    report = OutboxReport()
   128	    for entry in entries:
   129	        tool = entry["tool"]
   130	        planned = _request_for(tool, entry["args"], who)
   131	        if isinstance(planned, str):
   132	            report.dropped.append(planned)
   133	            continue
   134	        method, path, body = planned
   135	        try:
   136	            result = await call(method, path, body)
   137	        except Exception as exc:   # noqa: BLE001 - one bad entry must not lose the turn's other entries
   138	            report.dropped.append(f"{tool}: {type(exc).__name__}: {str(exc)[:120]}")
   139	            continue
   140	        error = result.get("error") if isinstance(result, dict) else None
   141	        if tool == "send_message":
   142	            if report_receipt_accepted(result):
   143	                error = None  # WAITING_QUOTA's error metadata describes a retained receipt.
   144	            else:
   145	                error = error or "parent report was not accepted"
   146	        if error:
   147	            report.dropped.append(f"{tool}: {str(error)[:160]}")
   148	        else:
   149	            report.done.append(tool)
   150	    for reason in report.dropped:
   151	        logger.warning("hub outbox of %s: dropped %s", who.name, reason)
   152	    return report
