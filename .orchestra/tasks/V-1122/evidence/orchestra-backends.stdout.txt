   766	                     stream: HubActivityStream) -> AsyncIterator[AgentEvent]:
   767	        """Wait for the run, translating new timeline items into events every poll slice."""
   768	        limit = float(os.environ.get("MULTICODE_HUB_TURN_TIMEOUT_S", "14400"))
   769	        deadline = time.monotonic() + limit
   770	        while True:
   771	            waited = await self._watch_call(mc, "t3_thread_wait", {
   772	                "threadId": thread_id, "runId": run_id, "timeoutMs": WAIT_SLICE_MS})
   773	            status = waited.get("status", "") if isinstance(waited, dict) else ""
   774	            status_limit = _provider_limit(status)
   775	            terminal = not (isinstance(waited, dict) and waited.get("timedOut")) and (status in TERMINAL or status_limit is not None)
   776	            for event in await self._poll(mc, thread_id, stream):
   777	                yield event
   778	            if terminal:
   779	                if self._run_evidence is not None:
   780	                    self._run_evidence.failure(waited)
   781	                self._watch_limit = self._watch_limit or status_limit or _provider_limit(waited)
   782	                self._watch_status = status
   783	                return
   784	            if self._abort:
   785	                self._watch_status = "interrupted"
   845	                self._run_evidence.page(read)
   846	            for run in (read.get("recentRuns") or []) + (read.get("runs") or []):
   847	                if isinstance(run, dict) and (run.get("id") or run.get("runId")) == run_id:
   848	                    self._watch_limit = self._watch_limit or _provider_limit(run)
   849	        return _last_assistant_text(read, run_id) or ""

   851	    async def _sync_back(self, ssh: hw.HubSsh, state: hw.WorkspaceState, root: Path,
   852	                         scratch: Path) -> str:
   853	        await hw.check_workspace_file_policy(ssh, state)
   854	        out = scratch / "out.tar"
   855	        await ssh.get(state.remote_dir, out)
   856	        before = dict(state.manifest)
   857	        hw.capture_inputs(root, state, state.manifest)
   858	        result = await asyncio.to_thread(hw.apply_changes, out, root, state.manifest,
   859	                                        inputs=state.input_manifest)
   860	        if any(hw._company_path(rel) for rel, _ in result.rejected):
   861	            # The snapshot is new evidence, including if someone later removes
   862	            # the file. Preserve it as a latch without rewriting old provenance.
   863	            state.file_policy_blocked = True
   864	        no_work = not (result.changed or result.deleted or result.conflicts
   865	                       or result.rejected or result.outbox or result.manifest != before)
   866	        state.manifest = result.manifest
   867	        state.hub_made = sorted((set(state.hub_made) | {r for r, d in result.manifest.items()
   868	                                                         if before.get(r) != d}) & set(result.manifest))
   869	        state.turns += 1
   870	        outbox_line = await self._run_outbox(ssh, state, result.outbox)
   871	        commit, unstaged = "", []
   872	        if result.changed or result.deleted:
   873	            message = f"{_branch_task_ref(self.cwd)}: hub worker turn {state.turns} ({self.model})"
   874	            try:
   875	                commit, unstaged = await asyncio.to_thread(
   876	                    hw.commit_sync, root, result.changed, result.deleted, message)
   877	            except (hw.HubWorkspaceError, subprocess.SubprocessError) as exc:
   878	                commit = f"NOT COMMITTED ({exc})"
   879	        state.save()
   880	        lines = [f"[hub sync] changed {len(result.changed)}, deleted {len(result.deleted)}"
   881	                 + (f", commit {commit}" if commit else "")]
   882	        lines.append(f"return archive {out.stat().st_size} bytes; "
   883	                     f"transport limit {hw.MAX_TOTAL_BYTES * 2} bytes; workspace {state.ws_id}")
   884	        if outbox_line:
   885	            lines.append(outbox_line)
   886	        if unstaged:
   887	            lines.append("left uncommitted in the worktree (git refused them): "
   888	                         + "; ".join(f"{p} ({why})" for p, why in unstaged[:20]))
   889	        if result.conflicts:
   890	            lines.append("conflicts (both sides changed, hub copy NOT applied): "
   891	                         + ", ".join(result.conflicts[:20]))
   892	        if result.rejected:
   893	            lines.append("rejected: " + "; ".join(f"{p} ({why})" for p, why in result.rejected[:20]))
   894	        self._sync_no_work = no_work
   895	        return "\n".join(lines)

   897	    async def _run_outbox(self, ssh: hw.HubSsh, state: hw.WorkspaceState, data: bytes) -> str:
   898	        """Run what the hub agent queued for Orchestra, then empty the file on the hub."""
   899	        if not data.strip():
   900	            return ""
   901	        entries, dropped = parse_outbox(data)
   902	        report = OutboxReport(dropped=dropped)
   903	        try:
   904	            await ssh.delete_files(state.remote_dir, [OUTBOX_NAME])   # each entry runs at most once
   905	        except hw.HubWorkspaceError as exc:
   906	            report.dropped.append(f"outbox not cleared on the hub, entries skipped: {exc}")
   907	            entries = []
   908	        if entries and self._bridge is None:
   909	            report.dropped.append(f"{len(entries)} entries: this worker has no Orchestra link")
   910	        elif entries:
   911	            ran = await self._bridge.run(entries)
   912	            report.done, report.dropped = ran.done, report.dropped + ran.dropped
   913	        for reason in report.dropped:
   914	            logger.warning("hub outbox of %s: dropped %s", state.ws_id, reason)
   915	        return report.summary()
