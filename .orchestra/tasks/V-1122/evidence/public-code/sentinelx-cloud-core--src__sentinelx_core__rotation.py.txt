     1	"""Agent-side credential rotation (phase 2).
     2	
     3	Every host's session credential expires 365 days after it was issued and
     4	nothing renewed it, so the whole fleet would age out (Aug-Sep 2027 for most).
     5	This rotates the credential past its half-life, which also lets the hub revoke
     6	a stolen copy at the next rotation.
     7	
     8	The one rule that governs every choice here: a rotation that goes wrong must
     9	NEVER leave a host unable to connect.
    10	
    11	  - The rotated credential is written to a SEPARATE file
    12	    (identity.rotated.json) in a directory the agent owns. identity.json --
    13	    root-owned, the original enrolment credential -- is never touched.
    14	  - On startup the agent prefers the rotated file IF it is present and valid,
    15	    and otherwise falls back to identity.json, which is still a valid
    16	    credential. A corrupt or half-written rotated file simply means "use the
    17	    original".
    18	  - The write is atomic: temp file in the same dir, fsync, os.replace. A crash
    19	    mid-write leaves either the old rotated file or none -- never a truncated one.
    20	  - If no writable directory exists, rotation is disabled with a warning and
    21	    the host keeps running on its current credential (covered by the hub's
    22	    legacy grace). Never worse than today.
    23	"""
    24	
    25	from __future__ import annotations
    26	
    27	import base64
    28	import json
    29	import os
    30	import tempfile
    31	import time
    32	from pathlib import Path
    33	
    34	from sentinelx_core.identity import Identity
    35	import logging
    36	
    37	logger = logging.getLogger(__name__)
    38	
    39	ROTATED_NAME = "identity.rotated.json"
    40	
    41	# Candidate directories the agent can write to, most-preferred first. The first
    42	# one that exists and is writable wins; identity.json's own dir is intentionally
    43	# NOT here (root-owned on Linux by design).
    44	_DIR_CANDIDATES = ("/var/lib/sentinelx", None)  # None -> alongside identity.json (Windows)
    45	
    46	
    47	def rotated_path(identity_path: Path) -> Path | None:
    48	    """Where the rotated credential lives, or None if nowhere is writable."""
    49	    for cand in _DIR_CANDIDATES:
    50	        d = Path(cand) if cand else identity_path.parent
    51	        try:
    52	            if d.is_dir() and os.access(d, os.W_OK):
    53	                return d / ROTATED_NAME
    54	        except OSError:
    55	            continue
    56	    return None
    57	
    58	
    59	def _claims(token: str) -> dict:
    60	    """Decode a JWT payload without verifying. It is our own credential; we only
    61	    read exp/iat to decide WHEN to rotate. The hub verifies on use."""
    62	    try:
    63	        seg = token.split(".")[1]
    64	        seg += "=" * (-len(seg) % 4)
    65	        return json.loads(base64.urlsafe_b64decode(seg))
    66	    except Exception:  # noqa: BLE001
    67	        return {}
    68	
    69	
    70	def load_effective_identity(identity_path: Path, base: Identity) -> Identity:
    71	    """The credential to connect with: the rotated one if usable, else the base.
    72	
    73	    `base` is whatever load_identity() returned from identity.json. This only
    74	    ever UPGRADES to a rotated credential that parses and is not expired; any
    75	    doubt falls back to base.
    76	    """
    77	    rp = rotated_path(identity_path)
    78	    if rp is None or not rp.exists():
    79	        return base
    80	    try:
    81	        data = json.loads(rp.read_text())
    82	        token = str(data["token"]).strip()
    83	        host_id = str(data.get("host_id") or base.host_id).strip()
    84	    except (OSError, ValueError, KeyError):
    85	        logger.warning("rotated_identity_unreadable; using original")
    86	        return base
    87	    exp = _claims(token).get("exp")
    88	    if not isinstance(exp, (int, float)) or exp <= time.time():
    89	        logger.warning("rotated_identity_expired_or_bad; using original")
    90	        return base
    91	    # Same host, and it must belong to the same account as the base credential.
    92	    if host_id != base.host_id:
    93	        logger.warning("rotated_identity_host_mismatch; using original")
    94	        return base
    95	    return Identity(host_id=host_id, token=token, hub=base.hub)
    96	
    97	
    98	def should_rotate(token: str) -> bool:
    99	    """True once the credential is past the midpoint of its lifetime.
   100	
   101	    Legacy tokens (no iat) are rotated too -- issued >182 days ago they are past
   102	    any 365-day midpoint, and rotating migrates them onto a typed, revocable
   103	    credential. A token we cannot read is left alone (return False): the hub
   104	    will reject it on connect if it is truly bad, which is the right place.
   105	    """
   106	    c = _claims(token)
   107	    exp, iat = c.get("exp"), c.get("iat")
   108	    if not isinstance(exp, (int, float)):
   109	        return False
   110	    if isinstance(iat, (int, float)):
   111	        return time.time() >= iat + (exp - iat) / 2
   112	    # No iat (legacy): rotate when less than half of a 365-day life remains.
   113	    return exp - time.time() < 182 * 86400
   114	
   115	
   116	def persist_rotated(identity_path: Path, host_id: str, token: str, hub: str) -> bool:
   117	    """Atomically write the rotated credential. Returns False if it could not be
   118	    written -- caller keeps using the current credential, no harm done."""
   119	    rp = rotated_path(identity_path)
   120	    if rp is None:
   121	        logger.warning("no writable dir for rotated credential; rotation disabled")
   122	        return False
   123	    payload = json.dumps({"host_id": host_id, "token": token, "hub": hub}, indent=2)
   124	    try:
   125	        fd, tmp = tempfile.mkstemp(dir=str(rp.parent), prefix=".idrot.")
   126	        try:
   127	            with os.fdopen(fd, "w") as fh:
   128	                fh.write(payload)
   129	                fh.flush()
   130	                os.fsync(fh.fileno())
   131	            os.chmod(tmp, 0o600)
   132	            os.replace(tmp, rp)  # atomic on POSIX and Windows
   133	        finally:
   134	            if os.path.exists(tmp):
   135	                os.unlink(tmp)
   136	        return True
   137	    except OSError as exc:
   138	        logger.warning("could not persist rotated credential: %s", exc)
   139	        return False
