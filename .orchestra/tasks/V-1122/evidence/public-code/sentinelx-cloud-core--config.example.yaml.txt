    50	# ============================================================================
    51	# How `exec` decides what may run. Three modes:
    52	#
    53	#   exec_mode: allowlist     (the default)
    54	#     Every command in the line must be covered by allowed_commands below, not
    55	#     just the first: the line is split on ; && || | & and newlines (outside
    56	#     quotes), and commands inside $(...), backticks and <( ) are checked too.
    57	#     Entries match whole words: `ls` allows `ls -la`, not `lsof`. Free in any
    58	#     position: cd, shell grammar (if/then/else/fi, for/do/done, while...),
    59	#     harmless builtins (set, test, true, read...), and read-only filters after
    60	#     the first command (head, grep, sort, cut, wc...) when they get no file and
    61	#     no write option. `VAR=value cmd` must match an entry as written.
    62	#     bash -lc '...' is one argument and isn't looked into: allowing bash (or
    63	#     sh, python3, php, node) allows anything.
    64	#
    65	#   exec_mode: strict
    66	#     As allowlist, and newlines, $(...)/backticks and file redirections are
    67	#     refused outright. This is what `exec_strict: true` meant; it still works.
    68	#
    69	#   exec_mode: unrestricted
    70	#     For trusted development hosts: exec runs any command line under the
    71	#     agent's identity, as script_run does, and allowed_commands is ignored for
    72	#     exec. The boundary is the operating system (the agent's user, sudoers,
    73	#     file permissions) and the audit log. Keep allowlist on production.
    74	#
    75	# disabled_ops: [exec] turns exec off whatever the mode.
    76	# ============================================================================
    77	exec_mode: allowlist

   246	  # A systemd USER unit (Linux, ~/.config/systemd/user): an app that runs as
   247	  # its own user, without root. With `user:` the agent talks to that user's
   248	  # manager (`systemctl --user`). If the agent itself runs as that user, no
   249	  # privileges are needed. Otherwise it uses
   250	  # `sudo systemctl --user --machine=<user>@.host <action> <unit>`, which needs
   251	  # root (a sudoers rule for exactly those commands) and systemd 248+.
   252	  # myapp.service:
   253	  #   user: alice
   254	  #   actions: [status, restart]
   255	  # The agent itself, so the LLM can reload policy after editing this
   256	  # file. Restart re-reads /etc/sentinelx/config.yaml from scratch
   257	  # (no SIGHUP-style reload). Conservative actions:
   258	  #   status / is-active / is-enabled — pure inspection

   260	  # Deliberately NOT start/stop: bringing the agent up/down is an

   268	# ============================================================================
   269	# Playbooks the agent exposes to the LLM via `capabilities`.
   270	#
   271	# A playbook is a named, structured procedure the LLM can follow to perform
   272	# a multi-step task on this host. Unlike a single command in
   273	# `allowed_commands`, a playbook is a recipe: it documents the sequence of
   274	# tool calls, the prerequisites, and any caveats. The LLM reads this from
   275	# the capabilities response and follows the steps when the user's intent
   276	# matches the `when` clause.
   277	#
   278	# The shape is flexible (dict[str, dict[str, Any]]). We use these keys by
   279	# convention:
   280	#   description: one-line summary
   281	#   when:        natural-language trigger ("when should the LLM use this?")
   282	#   steps:       ordered list of instructions, each phrased so the LLM can
   283	#                act on them directly
   284	#   requires:    prerequisites the operator must have set up beforehand
   285	#   notes:       caveats, alternatives, security considerations
   286	#
   287	# The agent does NOT execute playbooks itself — they are pure documentation
   288	# for the LLM. The actual work happens through the regular tools
   289	# (sentinel_exec, sentinel_edit, sentinel_service).
   290	# ============================================================================

   793	  drop_share:
   794	    description: |
   795	      Use drop.pensa.ar as a temporary buffer for files when transferring
   796	      between hosts, sharing host-side files with the operator/LLM, or
   797	      replicating one file to many hosts. drop.pensa.ar is a self-hosted
   798	      file-drop service (https://github.com/pensados/drop-pensa) that
   799	      accepts uploads up to 50 MB and returns a public URL with
   800	      configurable expiry. Files NEVER pass through the LLM's context.
   801	
   802	    when: |
   803	      Use this playbook in any of these scenarios:
   804	
   805	        (1) HOST-TO-HOST TRANSFER: copy a file from one enrolled host to
   806	            another (e.g., move /var/log/nginx/error.log from orion to
   807	            atlas for offline analysis). Bytes never touch the LLM
   808	            context.
   809	
   810	        (2) SHARE WITH OPERATOR / LLM FOR REVIEW: the user asks Claude
   811	            to look at a file that lives on a host (logs, configs,
   812	            dumps). Uploading via drop and pasting the URL into the
   813	            chat lets Claude web_fetch it without paying the cost of
   814	            base64-inlining a multi-MB file.
   815	
   816	        (3) REPLICATE ONE FILE TO MANY HOSTS: if the same artifact must
   817	            land on 3+ hosts (a config snippet, a binary, a cert), drop
   818	            it once and use sentinel_upload_file with file_url N times.
   819	            One POST instead of N base64 encodes.
   820	
   821	        (4) USER WANTS TO PUSH A FILE TO A HOST: the operator uploads
   822	            via the drop web UI (https://drop.pensa.ar), pastes the URL
   823	            in chat, then sentinel_upload_file with file_url lands it
   824	            on the target host without any base64 routing through the
   825	            LLM.
   826	
   827	      DO NOT use this playbook when:
   828	
   829	        - The LLM is creating a NEW file (text, config, code) and
   830	          uploading it to ONE host. In that case, sentinel_upload_file
   831	          with content_base64 (or sentinel_edit with mode='write' for
   832	          text) is one call versus three, with the same actual cost
   833	          on the LLM side. Drop adds steps without saving anything.
   834	
   835	        - The file is already at its destination. Don't round-trip
   836	          through drop just because.
   837	
   838	    steps:
   839	      - "(SCENARIO 1 — HOST-TO-HOST TRANSFER): on the SOURCE host, run:
   840	         sentinel_exec command='curl -s -F file=@/PATH/TO/FILE
   841	         https://drop.pensa.ar/upload?expires_in=600'.
   842	         The response is JSON; parse out the .url field. Tip: pipe to
   843	         python3 to extract cleanly:
   844	         sentinel_exec command='curl -s -F file=@/PATH https://drop.pensa.ar/upload
   845	         | python3 -c \"import sys,json; print(json.load(sys.stdin)[\\\"url\\\"])\"'.
   846	         Then on the DESTINATION host, call sentinel_upload_file with
   847	         file_url=<that URL> and target_path=<where you want it>.
   848	         The destination agent fetches drop.pensa.ar directly. Verify
   849	         with the sha256 returned in both responses — they MUST match."
   850	
   851	      - "(SCENARIO 2 — SHARE WITH OPERATOR/LLM): on the host that has
   852	         the file, run: sentinel_exec command='curl -s -F file=@/PATH
   853	         https://drop.pensa.ar/upload?expires_in=3600 | python3 -c
   854	         \"import sys,json; print(json.load(sys.stdin)[\\\"url\\\"])\"'.
   855	         Show the resulting URL to the operator. The operator pastes

   861	      - "(SCENARIO 3 — ONE-TO-MANY REPLICATION): identical to Scenario 1,

  1130	# file_ops: filesystem primitives, unified r/rw path model.
  1131	#
  1132	# `file_ops.paths` is the security boundary for ALL filesystem tools:
  1133	#   - read-only:  sentinel_read / sentinel_list / sentinel_search
  1134	#   - mutating:   sentinel_edit  and
  1135	#                 sentinel_move / sentinel_copy / sentinel_delete /
  1136	#                 sentinel_chmod / sentinel_chown
  1137	#
  1138	# Each entry has a `path` (a directory PREFIX) and an `access` level:
  1139	#   - access: r   -> read/list/search only. Edit and the destructive
  1140	#                    ops are refused with path_not_allowed.
  1141	#   - access: rw  -> read AND mutate. Required for edit/move/copy/
  1142	#                    delete/chmod/chown.
  1143	# A request is permitted only if the requested path, after
  1144	# canonicalization (symlinks resolved, '..' collapsed), falls under an
  1145	# entry that grants the needed access. An unknown/missing access value
  1146	# degrades to `r` (never `rw`) — safe by default. An empty or missing
  1147	# list disables every filesystem tool (path_not_allowed for all).
  1148	#
  1149	# Most-permissive match wins: if a path is under both an `r` parent and
  1150	# an `rw` subtree entry, a write to the subtree is allowed.
  1151	#
  1152	# BACK-COMPAT: the legacy key `file_ops.allowed_read_paths:` (a bare
  1153	# list of strings) is still accepted and each entry is treated as
  1154	# `access: r`. If BOTH `paths:` and `allowed_read_paths:` are present,
  1155	# `paths:` wins and a deprecation warning is logged. Migrate at your
  1156	# leisure; nothing breaks. Mixing is not recommended.
  1157	#
  1158	# This is SEPARATE from allowed_commands (which gates exec).
  1159	#
  1160	# Why a conservative default: the agent runs unattended on a remote
  1161	# host. Read feeds file content to the model (prompt-injection
  1162	# surface); write/delete can destroy data. The default below grants
  1163	# READ for system/service administration and grants RW to NOTHING —
  1164	# you must consciously opt a subtree into `rw`.
  1165	# ============================================================================
  1166	file_ops:
  1167	  paths:
  1168	    - path: /etc          # system and service configuration
  1169	      access: r
  1170	    - path: /var/log      # logs
  1171	      access: r
  1172	    - path: /var/www      # web content, if any
  1173	      access: r
  1174	    - path: /opt          # installed applications
  1175	      access: r
  1176	    - path: /srv          # served data
  1177	      access: r

  1181	[omitted: upstream personal/example home path]
  1182	    #     access: rw
  1183	    # Granting rw on a broad path (e.g. /etc) lets the model rewrite

  1186	    # Need read in a personal directory? Add the SPECIFIC path with

  1201	# ─────────────────────────────────────────────────────────────────────────────
  1202	# local_apis  (optional)  —  sentinel_local_api
  1203	#
  1204	# Bridge to host-local tools that ALREADY speak a structured protocol, so the
  1205	# model receives data instead of text meant for a terminal. `exec` remains the
  1206	# right answer for plain-text CLIs; this does not replace it.
  1207	#
  1208	# Leave this block out entirely and nothing changes: the agent registers no
  1209	# handler and advertises no such capability.
  1210	#
  1211	# ┌───────────────────────────────────────────────────────────────────────────
  1212	# │ READ THIS BEFORE ENABLING THE DOCKER ENDPOINT
  1213	# │
  1214	# │ /var/run/docker.sock is root:docker 0660, so the agent's user must be in
  1215	# │ the `docker` group to reach it. THAT GROUP IS ROOT-EQUIVALENT: whoever can
  1216	# │ talk to that socket can start a privileged container with the host
  1217	# │ filesystem mounted and take over the machine. Adding the agent to it
  1218	# │ widens what a compromised agent could do, no matter how tight
  1219	# │ `allowed_commands` is.
  1220	# │
  1221	# │   sudo usermod -aG docker sentinelx
  1222	# │   sudo systemctl restart sentinelx-cloud-core
  1223	# │
  1224	# │ That is a deliberate security decision, not a configuration detail. If it
  1225	# │ is not acceptable on this host, leave the endpoint out. Without the group
  1226	# │ the endpoint still loads and every call returns endpoint_unreachable.
  1227	# │
  1228	# │ TO UNDO IT: deleting this block does NOT remove the group membership. The
  1229	# │ agent keeps root-equivalent reach over Docker until you take it away:
  1230	# │
  1231	# │   sudo gpasswd -d sentinelx docker
  1232	# │   sudo systemctl restart sentinelx-cloud-core
  1233	# └───────────────────────────────────────────────────────────────────────────
  1234	#
  1235	# TWO RULES WORTH UNDERSTANDING BEFORE WRITING YOUR OWN:
  1236	#
  1237	# 1. `actions` is the entire security boundary, and it is required. `exec` is
  1238	#    bounded by allowed_commands (exec_mode), but a socket bridge is bounded only by what
  1239	#    the far side exposes, which the agent cannot see or enumerate. An endpoint
  1240	#    that declares no actions is NOT registered.
  1241	#
  1242	# 2. `select` is not cosmetic. Measured against nine real containers: the text
  1243	#    `docker ps` prints is 712 bytes, the raw socket JSON answering the same
  1244	#    question is 20,340, and the projection below is 1,811. Without `select`
  1245	#    the answer is worse than the text it replaces. Omit it only for endpoints
  1246	#    whose replies are already small.
  1247	#
  1248	# With transport `unix`, `path` is NOT a binary and actions are NOT
  1249	# subcommands: the agent opens a socket, so there is no process, no argv and
  1250	# no shell. The whole shell-injection class cannot occur on this path. (With
  1251	# transport `stdio`, at the end of this section, `path` is a binary you
  1252	# declare, still without a shell.)
  1253	#
  1254	# ON `run_as`. Some endpoints create their socket 0600 and check who is
  1255	# connecting, so reaching one means BEING its owner, not merely having
  1256	# permission to open it. Running the agent as root does not solve that: root
  1257	# connects but identifies as uid 0, which is not the owner.
  1258	#
  1259	# When `run_as` is set, the connect happens in a tiny relay invoked through
  1260	# `sudo -n -u <user>`. Since the agent is not granted sudo by default, the
  1261	# host's owner has to allow exactly that and nothing else:
  1262	#
  1263	#   sentinelx ALL=(youruser) NOPASSWD: /path/to/.venv/bin/python -m sentinelx_core.local_api_relay
  1264	#
  1265	# Until that rule exists the endpoint answers run_as_not_permitted and names
  1266	# the line to add. The relay takes a socket path and a timeout, writes one
  1267	# request and reads one reply: it loads no config, interprets no payload and
  1268	# runs nothing, and the path comes from this file rather than from the caller.
  1269	#
  1270	# local_apis:
  1271	#   docker:
  1272	#     transport: unix                  # unix | stdio
  1273	#     path: /var/run/docker.sock
  1274	#     protocol: http                   # http | jsonrpc
  1275	#     timeout_s: 20
  1276	#     actions:
  1277	#       # Reads only. The socket is root-equivalent, so nothing here starts,
  1278	#       # stops, removes or executes.
  1279	#       containers.list:
  1280	#         request: GET /v1.44/containers/json
  1281	#         select: [Names, Image, State, Status, Ports]
  1282	#         description: what is running on this host
  1283	#       containers.inspect:
  1284	#         request: GET /v1.44/containers/{id}/json    # {id} comes from params
  1285	#         select: [Name, State, Config.Image, HostConfig.RestartPolicy]
  1286	#         description: full detail for one container
  1287	#       images.list:
  1288	#         request: GET /v1.44/images/json
  1289	#         select: [RepoTags, Size, Created]
  1290	#
  1291	#   # A JSON-RPC endpoint looks like this instead. Actions name methods.
  1292	#   # some_tool:
  1293	#   #   transport: unix
  1294	#   #   path: /run/user/1000/some_tool.sock
  1295	#   #   protocol: jsonrpc
  1296	#   #   run_as: youruser                # connect AS this Unix user
  1297	#   #   compatibility:
  1298	#   #     probe:   { method: session.describe }  # how to ASK for the version
  1299	#   #     extract: protocol                      # where to read it in the reply
  1300	#   #     accept:  { exact: 20 }                 # or { allowed: [20, 21] }
  1301	#   #   actions:
  1302	#   #     session.snapshot: { method: session.snapshot }
  1303	#   #     # `params:` on an action declares its parameter SHAPE, not values.
  1304	#   #     # An HTTP action does not need it: the names are already readable
  1305	#   #     # from the {placeholders} in its request. A JSON-RPC action has no
  1306	#   #     # template, so without this a caller is told the action exists and
  1307	#   #     # nothing about what it takes.
  1308	#   #     #
  1309	#   #     # Carried verbatim and never interpreted here, so nesting, arrays
  1310	#   #     # and enums mean whatever the endpoint says they mean.
  1311	#   #     agent.wait:
  1312	#   #       method: agent.wait
  1313	#   #       params:
  1314	#   #         type: object
  1315	#   #         required: [target]
  1316	#   #         properties:
  1317	#   #           target: { type: string }
  1318	#   #           until:
  1319	#   #             type: array
  1320	#   #             items:
  1321	#   #               enum: [idle, done, blocked]
  1322	#   #           timeout_ms:
  1323	#   #             type: [integer, "null"]
  1324	#
  1325	# On `compatibility`, both halves are declared: the profile says how to obtain
  1326	# the metadata and which values it accepts, and SentinelX evaluates that
  1327	# declared constraint. It is NEVER inferred from `new_version >= configured`,
  1328	# because a higher number does not imply the protocol still matches; `accept.
  1329	# allowed` is how whoever does know widens it deliberately.
  1330	#
  1331	# It is checked once per connection epoch, not per call: probing before every
  1332	# call pays a round trip for a guarantee it cannot give, since the endpoint can
  1333	# restart between the check and the call either way. The verdict is dropped the
  1334	# moment the endpoint becomes unreachable, so an endpoint that comes back
  1335	# speaking something else is judged again rather than inheriting the old answer.
  1336	#
  1337	# It fails CLOSED. An endpoint that cannot be probed, or that does not report
  1338	# the field named in `extract`, is refused rather than assumed fine. A block
  1339	# missing any of the three keys is dropped with a warning instead of being
  1340	# half-enforced, because a constraint that silently does nothing reads as
  1341	# protection that is not there.
  1342	#
  1343	# ── transport: stdio  —  a binary that speaks JSON-RPC on stdin/stdout ──────
  1344	#
  1345	# For tools that run as a process instead of listening on a socket: MCP
  1346	# servers launched over stdio, language servers, small JSON-RPC helpers. It is
  1347	# also the transport that works on Windows, which has no unix sockets.
  1348	#
  1349	# Here `path` IS a binary, and you declare it. The agent starts it once per
  1350	# call, writes the JSON-RPC request as one line, reads stdout until the answer
  1351	# with the same id, then stops it: stdin closed, and whatever is still running
  1352	# after a short grace killed as a whole process tree. There is no shell: argv
  1353	# is `path` plus the FIXED `args`, and a call's parameters travel only inside
  1354	# the JSON message, never as arguments, flags or file names.
  1355	#
  1356	#   protocol   must be jsonrpc.
  1357	#   path       absolute. The binary, and the directories holding it, must not
  1358	#              be writable by anyone but their owner: whoever can replace the
  1359	#              file chooses what the agent runs. Such an endpoint is skipped at
  1360	#              load and refused at call time (unsafe_binary). A binary that does
  1361	#              not exist yet still registers; calls fail until it is installed.
  1362	#   args       optional, a fixed list. `describe` shows how many, not what they
  1363	#              are, so a token passed this way is not echoed to the model.
  1364	#   cwd        optional, absolute.
  1365	#   env        optional variables, added to a MINIMAL environment (PATH, HOME,
  1366	#              locale; the system variables on Windows). The agent's own
  1367	#              environment is not passed on.
  1368	#   handshake  none (default) | mcp. With mcp the agent sends `initialize` and
  1369	#              `notifications/initialized` first, as an MCP server requires.
  1370	#   timeout_s  the whole call: start-up and handshake included.
  1371	#   run_as     starts it as `sudo -n -u <user> -- <path> <args>`. The sudoers
  1372	#              rule names that exact command and nothing wider, e.g.
  1373	#                sentinelx ALL=(youruser) NOPASSWD: /opt/tools/bin/inventory --stdio
  1374	#              `env` cannot be combined with run_as: sudo would drop it.
  1375	#
  1376	# ON MCP SERVERS. The allowlist is of METHODS. An action with
  1377	# `method: tools/call` reaches EVERY tool the server offers, whatever its
  1378	# `params` schema says: the schema is shown to the caller, not enforced. Pick
  1379	# servers whose whole tool set you accept, or limit their tools with the
  1380	# server's own options in `args`.
  1381	#
  1382	#   # inventory:
  1383	#   #   transport: stdio
  1384	#   #   protocol: jsonrpc
  1385	#   #   path: /opt/tools/bin/inventory
  1386	#   #   args: ["--stdio"]
  1387	#   #   timeout_s: 15
  1388	#   #   actions:
  1389	#   #     hosts.list: { method: hosts.list }
  1390	#   #
  1391	#   # An MCP server over stdio, declared the way an MCP client declares it:
  1392	#   # files:
  1393	#   #   transport: stdio
  1394	#   #   protocol: jsonrpc
  1395	#   #   path: /usr/bin/node              # Windows: 'C:\Program Files\nodejs\node.exe'
  1396	#   #   args: ["/opt/mcp/server-filesystem/dist/index.js", "/srv/shared"]
  1397	#   #   handshake: mcp
  1398	#   #   timeout_s: 30
  1399	#   #   actions:
  1400	#   #     tools.list: { method: tools/list }
  1401	#   #     tools.call:
  1402	#   #       method: tools/call             # params: { name, arguments }
  1403	#   #       params:
  1404	#   #         type: object
  1405	#   #         required: [name]
  1406	#   #         properties:
  1407	#   #           name: { type: string }
  1408	#   #           arguments: { type: object }
