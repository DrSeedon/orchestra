def digest(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()

    if result.returncode:
        raise RuntimeError(f"command failed ({result.returncode}): {' '.join(command)}\n{result.stderr.decode(errors='replace')}")
    return result
